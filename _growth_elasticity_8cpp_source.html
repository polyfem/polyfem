27 template <typename T>
28 void get_local_state(
29 const MixedNonLinearAssemblerData &data,
30 const int dim,
31 Eigen::Matrix<T, Eigen::Dynamic, 1> &local_state)
32 {
33 const int n_phi_bases = int(data.phi_vals.basis_values.size());
34 const int n_psi_bases = int(data.psi_vals.basis_values.size());
35 const int phi_local_size = n_phi_bases * dim;
36 const int local_size = phi_local_size + n_psi_bases;
37
38 Eigen::VectorXd values = Eigen::VectorXd::Zero(local_size);
39 for (int i = 0; i < n_phi_bases; ++i)
40 {
41 const auto &bs = data.phi_vals.basis_values[i];
42 for (const auto &global : bs.global)
43 {
44 for (int d = 0; d < dim; ++d)
45 values(i * dim + d) += global.val * data.x_phi(global.index * dim + d);
46 }
47 }
48
49 for (int i = 0; i < n_psi_bases; ++i)
50 {
51 const auto &bs = data.psi_vals.basis_values[i];
52 for (const auto &global : bs.global)
53 values(phi_local_size + i) += global.val * data.x_psi(global.index);
54 }
55
56 DiffScalarBase::setVariableCount(local_size);
57 local_state.resize(local_size);
58
59 const AutoDiffAllocator<T> allocate_auto_diff_scalar;
60 for (int i = 0; i < local_size; ++i)
61 local_state(i) = allocate_auto_diff_scalar(i, values(i));
62 }

64 template <typename T>
65 void displacement_gradient_at_quad(
66 const MixedNonLinearAssemblerData &data,
67 const Eigen::Matrix<T, Eigen::Dynamic, 1> &local_state,
68 const int p,
69 const int dim,
70 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> &grad_u)
71 {
72 grad_u.resize(dim, dim);
73 for (int k = 0; k < grad_u.size(); ++k)
74 grad_u(k) = T(0);
75
76 for (int i = 0; i < data.phi_vals.basis_values.size(); ++i)
77 {
78 const auto &bs = data.phi_vals.basis_values[i];
79 const Eigen::Matrix<double, Eigen::Dynamic, 1, 0, 3, 1> grad = bs.grad.row(p);
80 assert(grad.size() == dim);
81
82 for (int d = 0; d < dim; ++d)
83 {
84 for (int c = 0; c < dim; ++c)
85 grad_u(d, c) += grad(c) * local_state(i * dim + d);
86 }
87 }
88
89 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> jac_it(dim, dim);
90 for (int k = 0; k < jac_it.size(); ++k)
91 jac_it(k) = T(data.phi_vals.jac_it[p](k));
92 grad_u = grad_u * jac_it;
93 }

95 template <typename T>
96 T growth_at_quad(
97 const MixedNonLinearAssemblerData &data,
98 const Eigen::Matrix<T, Eigen::Dynamic, 1> &local_state,
99 const int p,
100 const int dim)
101 {
102 const int phi_local_size = int(data.phi_vals.basis_values.size()) * dim;
103 T theta = T(0);
104 for (int i = 0; i < data.psi_vals.basis_values.size(); ++i)
105 theta += data.psi_vals.basis_values[i].val(p) * local_state(phi_local_size + i);
106 return theta;
107 }

113 // ------------------------------------------------------------------
114 // Per-element vector/scalar cell data.
115 //
116 // Minimal reader for legacy ASCII VTK cell data: CELL_DATA section,
117 // arrays as "VECTORS <name> ...", "SCALARS <name> ..." or inside
118 // "FIELD FieldData" as "<name> <ncomp> <ntuples> <dtype>". (The fiber
119 // reader in MatParams.cpp handles VECTORS only.)
120 // ------------------------------------------------------------------
121 Eigen::MatrixXd read_vtk_cell_array(
122 std::istream &in, const std::string &path, const std::string &field, const int ncomp_expected)
123 {
124 if (!in)
125 log_and_throw_error("GrowthElasticity: cannot open '{}'", path);
126
127 std::string line;
128 bool in_cell_data = false;
129 long n_cells = -1;
130
131 while (std::getline(in, line))
132 {
133 std::istringstream ls(line);
134 std::string tok;
135 ls >> tok;
136
137 if (tok == "CELL_DATA")
138 {
139 in_cell_data = true;
140 ls >> n_cells;
141 continue;
142 }
143 if (tok == "POINT_DATA")
144 {
145 in_cell_data = false;
146 continue;
147 }
148 if (!in_cell_data)
149 continue;
150
151 long ntuples = n_cells;
152 int ncomp = -1;
153 std::string name;
154
155 if (tok == "VECTORS")
156 {
157 ls >> name;
158 ncomp = 3;
159 }
160 else if (tok == "FIELD")
161 {
162 continue; // array headers follow on their own lines
163 }
164 else if (tok == "SCALARS")
165 {
166 ls >> name;
167 ncomp = 1;
168 std::getline(in, line); // LOOKUP_TABLE line
169 }
170 else
171 {
172 // possible FIELD array header: <name> <ncomp> <ntuples> <dtype>
173 name = tok;
174 if (!(ls >> ncomp >> ntuples))
175 continue;
176 }
177
178 if (name != field)
179 {
180 // skip this array's payload
181 long remaining = ntuples * std::max(ncomp, 1);
182 double dummy;
183 while (remaining > 0 && in >> dummy)
184 --remaining;
185 continue;
186 }
187
188 if (ncomp != ncomp_expected)
189 log_and_throw_error(
190 "GrowthElasticity: field '{}' in '{}' has {} components, expected {}",
191 field, path, ncomp, ncomp_expected);
192
193 Eigen::MatrixXd values(ntuples, ncomp);
194 for (long i = 0; i < ntuples; ++i)
195 for (int c = 0; c < ncomp; ++c)
196 if (!(in >> values(i, c)))
197 log_and_throw_error(
198 "GrowthElasticity: truncated field '{}' in '{}'", field, path);
199 return values;
200 }
201
202 log_and_throw_error(
203 "GrowthElasticity: cell field '{}' not found in '{}'", field, path);
204 return Eigen::MatrixXd(); // unreachable
205 }

308 template <typename T>
309 T compute_energy_aux(const MixedNonLinearAssemblerData &data) const
310 {
311 if (size_ != 3)
312 log_and_throw_error("GrowthElasticity supports 3D only (in-plane growth needs a wall normal).");
313 assert(data.phi_vals.basis_values.size() > 0);
314 assert(data.psi_vals.basis_values.size() > 0);
315 assert(data.phi_vals.quadrature.weights.size() == data.psi_vals.quadrature.weights.size());
316
317 const int dim = size_;
318 const int el_id = data.phi_vals.element_id;
319
320 Eigen::Matrix<T, Eigen::Dynamic, 1> local_state;
321 get_local_state(data, dim, local_state);
322
323 const Eigen::Vector3d n0v = n0(el_id);
324 const double vn_e = vn(el_id);
325 if (!(vn_e > 0))
326 log_and_throw_error("GrowthElasticity: normal_growth must be positive, got {} on element {}.", vn_e, el_id);
327
328 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> grad_u(dim, dim);
329 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> F(dim, dim), Fgi(dim, dim), Fe(dim, dim);
330
331 T energy = T(0);
332 for (int p = 0; p < data.da.size(); ++p)
333 {
334 displacement_gradient_at_quad(data, local_state, p, dim, grad_u);
335
336 F = grad_u;
337 for (int d = 0; d < dim; ++d)
338 F(d, d) += T(1);
339
340 const T theta = growth_at_quad(data, local_state, p, dim);
341 if (!(value_of(theta) > 0))
342 log_and_throw_error(
343 "GrowthElasticity: growth theta must stay positive; got {} on element {} "
344 "(the growth-block feasibility filter is responsible for keeping iterates admissible).",
345 value_of(theta), el_id);
346
347 // Fg^-1 = I + (theta^{-1/2} - 1)(I - n0 x n0) + (vn^{-1} - 1) n0 x n0.
348 // Identity-anchored form: at theta == 1, vn == 1 both coefficients
349 // are exactly zero, so Fgi is the BITWISE identity for any normal
350 // and the correction vanishes exactly.
351 using std::sqrt;
352 const T c_ip = T(1) / sqrt(theta) - T(1);
353 const double c_n = 1.0 / vn_e - 1.0;
354 for (int a = 0; a < dim; ++a)
355 for (int b = 0; b < dim; ++b)
356 {
357 const double id = (a == b) ? 1.0 : 0.0;
358 const double nn = n0v(a) * n0v(b);
359 Fgi(a, b) = T(id) + c_ip * T(id - nn) + T(c_n * nn);
360 }
361
362 Fe = F * Fgi;
363 const T Jg = theta * T(vn_e);
364
365 // energy per intermediate volume: J^g * psi_e(Fe),
366 // minus psi_e(F) so this form is the additive correction and
367 // vanishes identically at theta = 1, vn = 1.
368 const RowVectorNd pos = data.phi_vals.val.row(p);
369 energy += (Jg * sum_energy<T>(pos, data.t, el_id, Fe)
370 - sum_energy<T>(pos, data.t, el_id, F))
371 * data.da(p);
372 }
373
374 return energy;
375 }

377 double driving_force(const int el_id, const RowVectorNd &pos, const double t, const Eigen::Matrix3d &Ce) const
378 {
379 if (size_ != 3)
380 log_and_throw_error("GrowthElasticity supports 3D only.");
381 const Eigen::SelfAdjointEigenSolver<Eigen::Matrix3d> es(0.5 * (Ce + Ce.transpose()));
382 if (es.info() != Eigen::Success || !(es.eigenvalues().minCoeff() > 0))
383 log_and_throw_error("GrowthElasticity::driving_force: Ce is not positive definite on element {}.", el_id);
384 // f_I depends on Ce only (objectivity), so any Fe with Fe^T Fe = Ce
385 // will do; the symmetric root is the canonical choice.
386 const Eigen::Matrix3d Fe = es.eigenvectors() * es.eigenvalues().cwiseSqrt().asDiagonal() * es.eigenvectors().transpose();
387 const Eigen::Vector3d n0v = n0(el_id);
388
389 using AD = DScalar1<double, Eigen::VectorXd>;
390 DiffScalarBase::setVariableCount(1);
391 const AD theta(0, 1.0);
392 using std::sqrt;
393 const AD c_ip = AD(1) / sqrt(theta) - AD(1);
394 Eigen::Matrix<AD, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> Fs(3, 3);
395 for (int a = 0; a < 3; ++a)
396 for (int b = 0; b < 3; ++b)
397 {
398 // Fe * Fgi(theta), Fgi = I + c_ip (I - n0 n0), vn = 1
399 AD v = AD(Fe(a, b));
400 for (int k = 0; k < 3; ++k)
401 {
402 const double ip = (k == b ? 1.0 : 0.0) - n0v(k) * n0v(b);
403 v += AD(Fe(a, k) * ip) * c_ip;
404 }
405 Fs(a, b) = v;
406 }
407 const AD g = theta * sum_energy<AD>(pos, t, el_id, Fs);
408 return -g.getGradient()(0);
409 }

413 void grown_stresses(
414 const int el_id, const RowVectorNd &pos, const double t,
415 const Eigen::Matrix3d &F, const double theta,
416 Eigen::Matrix3d &cauchy, Eigen::Matrix3d &pk1, Eigen::Matrix3d &pk2,
417 double &energy) const
418 {
419 if (size_ != 3)
420 log_and_throw_error("GrowthElasticity supports 3D only.");
421 if (!(theta > 0))
422 log_and_throw_error("GrowthElasticity::grown_stresses: growth theta must be positive; got {} on element {}.", theta, el_id);
423 const double J = F.determinant();
424 if (!(J > 0))
425 log_and_throw_error("GrowthElasticity::grown_stresses: det F = {} on element {}.", J, el_id);
426
427 // Fg^-1 in the identity-anchored form of the energy, so theta == 1,
428 // vn == 1 gives Fe == F exactly.
429 const Eigen::Vector3d n0v = n0(el_id);
430 const double vn_e = vn(el_id);
431 const Eigen::Matrix3d nn = n0v * n0v.transpose();
432 const Eigen::Matrix3d I = Eigen::Matrix3d::Identity();
433 const Eigen::Matrix3d Fgi = I + (1.0 / std::sqrt(theta) - 1.0) * (I - nn) + (1.0 / vn_e - 1.0) * nn;
434 const Eigen::Matrix3d Fe = F * Fgi;
435 const double Jg = theta * vn_e;
436
437 // Pe = d psi_e / d Fe through the same templated child energies
438 using AD = DScalar1<double, Eigen::VectorXd>;
439 DiffScalarBase::setVariableCount(9);
440 Eigen::Matrix<AD, Eigen::Dynamic, Eigen::Dynamic, 0, 3, 3> Fe_ad(3, 3);
441 for (int a = 0; a < 3; ++a)
442 for (int b = 0; b < 3; ++b)
443 Fe_ad(a, b) = AD(a * 3 + b, Fe(a, b));
444 const AD psi = sum_energy<AD>(pos, t, el_id, Fe_ad);
445 Eigen::Matrix3d Pe;
446 for (int a = 0; a < 3; ++a)
447 for (int b = 0; b < 3; ++b)
448 Pe(a, b) = psi.getGradient()(a * 3 + b);
449
450 pk1 = Jg * Pe * Fgi.transpose();
451 cauchy = pk1 * F.transpose() / J;
452 pk2 = F.inverse() * pk1;
453 energy = Jg * psi.getValue();
454 }
