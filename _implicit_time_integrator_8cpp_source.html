103 Eigen::MatrixXd v = reader.read_matrix(group + "/v");
104 Eigen::MatrixXd a = reader.read_matrix(group + "/a");

141 else

144 throw std::runtime_error(fmt::format("Unknown time integrator ({})", type));
145 }
146
147 if (params.is_object())
148 integrator->set_parameters(params);
149

163 type);

165
166 auto integrator = std::make_shared<BDF>(

170 dynamic_order);
171 if (params.is_object() && params.contains("steps"))
