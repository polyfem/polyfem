25 // Logical paths are POSIX paths, regardless of the host operating system.
26 std::string normalize_logical(const std::string &path)
27 {
28 const bool absolute = !path.empty() && path.front() == '/';
29 std::vector<std::string> components;
30 std::istringstream input(path);
31 std::string component;
32 while (std::getline(input, component, '/'))
33 {
34 if (component.empty() || component == ".")
35 continue;
36 if (component == ".." && !components.empty() && components.back() != "..")
37 components.pop_back();
38 else if (component != ".." || !absolute)
39 components.push_back(component);
40 }
41 std::string result = absolute ? "/" : "";
42 for (const auto &part : components)
43 {
44 if (!result.empty() && result.back() != '/')
45 result += '/';
46 result += part;
47 }
48 return result.empty() ? "." : result;
49 }

77 std::string relative_logical(const std::string &path, const std::string &base)
78 {
79 const auto split = [](const std::string &value) {
80 std::vector<std::string> parts;
81 std::istringstream input(value);
82 std::string part;
83 while (std::getline(input, part, '/'))
84 if (!part.empty())
85 parts.push_back(part);
86 return parts;
87 };
88 const auto target = split(path), root = split(base);
89 size_t common = 0;
90 while (common < target.size() && common < root.size() && target[common] == root[common])
91 ++common;
92 std::string result;
93 for (size_t i = common; i < root.size(); ++i)
94 result += "../";
95 for (size_t i = common; i < target.size(); ++i)
96 result += target[i] + "/";
97 if (!result.empty())
98 result.pop_back();
99 return result.empty() ? "." : result;

128 template <typename T>
129 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> read_numeric_matrix(const h5pp::DsetInfo &info)
130 {
131 const std::string &path = info.dsetPath.value();
132 const auto &dims = info.dsetDims.value();
133 if (dims.size() > 2)
134 log_and_throw_error("Numeric resource {} must be a scalar, vector or matrix.", path);
135 const hsize_t rows = dims.empty() ? 1 : dims[0];
136 const hsize_t cols = dims.size() < 2 ? 1 : dims[1];
137 const hsize_t limit = std::numeric_limits<Eigen::Index>::max();
138 if (rows > limit || cols > limit || (cols && rows > limit / cols))
139 log_and_throw_error("Numeric resource {} is too large.", path);
140 Eigen::Matrix<T, Eigen::Dynamic, Eigen::Dynamic> result(rows, cols);
141 const hid_t type = info.h5Type.value();
142 const H5T_class_t kind = H5Tget_class(type);
143 const auto read = [&](auto scalar, const hid_t memory_type) {
144 using Source = decltype(scalar);
145 std::vector<Source> values(result.size());
146 if (!values.empty() && H5Dread(info.h5Dset.value(), memory_type, H5S_ALL, H5S_ALL, H5P_DEFAULT, values.data()) < 0)
147 log_and_throw_error("Unable to read numeric resource {}.", path);
148 for (Eigen::Index r = 0; r < result.rows(); ++r)
149 for (Eigen::Index c = 0; c < result.cols(); ++c)
150 {
151 const Source value = values[r * result.cols() + c];
152 if constexpr (std::is_integral_v<T> && std::is_integral_v<Source>)
153 {
154 if constexpr (std::is_signed_v<Source>)
155 {
156 if (value < std::numeric_limits<T>::lowest() || value > std::numeric_limits<T>::max())
157 log_and_throw_error("Integer value in {} is out of range for the requested type.", path);
158 }
159 else if (value > uint64_t(std::numeric_limits<T>::max()))
160 log_and_throw_error("Integer value in {} is out of range for the requested type.", path);
161 }
162 result(r, c) = static_cast<T>(value);
163 }
164 };
165 if (kind == H5T_INTEGER && H5Tget_size(type) <= sizeof(uint64_t))
166 {
167 if (H5Tget_sign(type) == H5T_SGN_NONE)
168 read(uint64_t(0), H5T_NATIVE_UINT64);
169 else
170 read(int64_t(0), H5T_NATIVE_INT64);
171 }
172 else if (kind == H5T_FLOAT && std::is_floating_point_v<T>)
173 read(double(0), H5T_NATIVE_DOUBLE);
174 else
175 log_and_throw_error("Numeric resource {} has an incompatible datatype.", path);
176 return result;
177 }

202 std::vector<std::string> ResourceIO::glob(const std::string &input_pattern) const
203 {
204 const std::string pattern = normalize_logical(input_pattern);
205 if (pattern.find_first_of("*?") == std::string::npos)
206 return exists(pattern) ? std::vector<std::string>{pattern} : std::vector<std::string>{};
207 const bool recursive = pattern.find("**") != std::string::npos;
208 const auto depth = std::count(pattern.begin(), pattern.end(), '/');
209 std::string expression;
210 expression.reserve(pattern.size() * 2);
211 for (size_t i = 0; i < pattern.size(); ++i)
212 {
213 const char c = pattern[i];
214 if (c == '*' && i + 1 < pattern.size() && pattern[i + 1] == '*')
215 {
216 ++i;
217 if (i + 1 < pattern.size() && pattern[i + 1] == '/')
218 {
219 expression += "(?:[^/]+/)*";
220 ++i;
221 }
222 else
223 expression += ".*";
224 }
225 else if (c == '*')
226 expression += "[^/]*";
227 else if (c == '?')
228 expression += "[^/]";
229 else
230 {
231 if (std::string(".^$|()[]{}+\\").find(c) != std::string::npos)
232 expression += '\\';
233 expression += c;
234 }
235 }
236 const std::regex matcher("^" + expression + "$");
237 const size_t wildcard = pattern.find_first_of("*?");
238 const std::string prefix = wildcard == std::string::npos ? pattern : pattern.substr(0, wildcard);
239 const size_t slash = prefix.rfind('/');
240 const std::string root = slash == std::string::npos ? "." : prefix.substr(0, slash);
241
242 std::vector<std::string> result;
243 std::vector<std::string> pending{root.empty() ? "/" : root};
244 while (!pending.empty())
245 {
246 const std::string current = pending.back();
247 pending.pop_back();
248 for (const std::string &child : list(current))
249 {
250 if (std::regex_match(child, matcher))
251 result.push_back(child);
252 if ((recursive || std::count(child.begin(), child.end(), '/') < depth) && is_group(child))
253 pending.push_back(child);
254 }
255 }
256 std::sort(result.begin(), result.end());
257 return result;
258 }

402 std::string HDF5IO::logical_resolve(const std::string &path) const
403 {
404 // Mounted checkpoint resources retain the original filesystem namespace.
405 // Ordinary HDF5 bundles still treat 'C:' and backslashes as POSIX names.
406 if (!storage_root_.empty() && is_windows_absolute(root_))
407 {

412 return normalize_windows_absolute(root_.substr(0, 2) + normalized);
413 return normalize_windows_absolute(root_ + "/" + normalized);
414 }
415 return join_logical(root_, path);

436 std::vector<std::string> HDF5IO::list(const std::string &path) const
437 {
438 if (!is_group(path))
439 return {};
440 const std::string parent = resolve(path);
441 const std::string logical_parent = logical_resolve(path);
442 const bool absolute_input = (!path.empty() && path.front() == '/')
443 || (!storage_root_.empty() && is_windows_absolute(root_) && is_windows_absolute(path));
444 std::set<std::string> children;
445 const auto add = [&](const std::vector<std::string> &entries) {
446 for (const std::string &entry : entries)
447 {
448 std::string full = entry;
449 if (full.empty())
450 continue;
451 if (full.front() != '/')
452 full = join_logical(parent, full);
453 const std::string relative = relative_logical(normalize_logical(full), parent);
454 if (relative != "." && relative != ".." && relative.compare(0, 3, "../") != 0)
455 {
456 const std::string child = join_logical(logical_parent, relative.substr(0, relative.find('/')));
457 children.insert(
458 absolute_input ? child : relative_logical(child, root_));
459 }
460 }
461 };
462 add(impl_->file.findGroups("", parent));
463 add(impl_->file.findDatasets("", parent));
464 return {children.begin(), children.end()};

541 fs::path HDF5IO::materialize(const std::string &path) const
542 {
543 record_access(path);
544 const std::string key = resolve(path);
545 std::lock_guard<std::mutex> lock(impl_->mutex);
546 if (const auto it = impl_->materialized.find(key); it != impl_->materialized.end())
547 return it->second;
548 if (impl_->temporary_directory.empty())
549 {
550 const auto stamp = std::chrono::high_resolution_clock::now().time_since_epoch().count();
551 for (size_t attempt = 0;; ++attempt)
552 {
553 const fs::path candidate = fs::temp_directory_path() / fmt::format("polyfem-resources-{}-{}", stamp, attempt);
554 if (fs::create_directory(candidate))
555 {
556 impl_->temporary_directory = candidate;
557 break;
558 }
559 }
560 }
561 // Separate directories preserve the original extension without aliasing
562 // resources that happen to have the same basename.
563 const fs::path directory = impl_->temporary_directory / std::to_string(impl_->materialized.size());
564 fs::create_directory(directory);
565 const fs::path filename = fs::path(key).filename();
566 const fs::path output = directory / (filename.empty() ? fs::path("resource") : filename);
567 std::ofstream file(output, std::ios::binary);
568 const std::string contents = read_string(path);
569 file.write(contents.data(), contents.size());
570 if (!file)
571 log_and_throw_error("Unable to materialize HDF5 resource {}", describe(path));
572 impl_->materialized.emplace(key, output);
573 return output;
574 }
