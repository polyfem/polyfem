29 double point_segment_squared_distance(const Vec3 &point, const Vec3 &v0, const Vec3 &v1)
30 {
31 const double l2 = distance2(v0, v1);
32 const double t = dot(point - v0, v1 - v0);
33 if (t <= 0.0 || l2 == 0.0)
34 return distance2(point, v0);
35 else if (t > l2)
36 return distance2(point, v1);
37 const double lambda1 = t / l2;
38 const double lambda0 = 1.0 - lambda1;
39 const Vec3 closest_point = lambda0 * v0 + lambda1 * v1;
40 return distance2(point, closest_point);
41 }
