477 void NonlinearElasticVarForm::build_periodic_collision_mesh()
478 {
479 const int dim = mesh_->dimension();
480 const int n_tiles = 2;
481 const int tile_count = 1 << dim;
482
483 if (obstacle.n_vertices() != 0)
484 {
485 log_and_throw_error("Periodic contact does not support obstacles.");
486 }
487
488 const int n_bases = space_.n_bases;
489 const json &conditions = args["boundary_conditions"]["periodic"];
490 Eigen::VectorXi periodic_dof_mask = Eigen::VectorXi::Zero(n_bases);
491 Eigen::VectorXi periodic_face_mask = Eigen::VectorXi::Zero(n_bases);
492 Eigen::MatrixXd periodic_tile_offsets(dim, conditions.size());
493 for (int i = 0; i < int(conditions.size()); ++i)
494 {
495 const json &condition = conditions[i];
496 const std::array<int, 2> boundary_ids = {{condition["boundary_ids"][0].get<int>(),
497 condition["boundary_ids"][1].get<int>()}};
498 const auto mapping = solver::PeriodicBoundaryLagrangianForm::build_mapping(
499 n_bases * dim, dim, *mesh_, space_.basis_list(), boundary_.total_local_boundary,
500 boundary_ids, condition.value("tolerance", 1e-5));
501 periodic_tile_offsets.col(i) = mapping.translation.transpose();
502 for (const int dof : mapping.boundary_dofs)
503 {
504 periodic_dof_mask(dof) = 1;
505 }
506 for (const auto &boundary : boundary_.total_local_boundary)
507 {
508 const auto &bases = space_.basis_list()[boundary.element_id()];
509 for (int primitive = 0; primitive < boundary.size(); ++primitive)
510 {
511 const int primitive_id = boundary.global_primitive_id(primitive);
512 for (int side = 0; side < 2; ++side)
513 {
514 if (mesh_->get_boundary_id(primitive_id) != boundary_ids[side])
515 {
516 continue;
517 }
518 for (const int node : bases.local_nodes_for_primitive(primitive_id, *mesh_))
519 {
520 for (const auto &global : bases.bases[node].global())
521 {
522 periodic_face_mask(global.index) |= 1 << (2 * i + side);
523 }
524 }
525 }
526 }
527 }
528 }
529
530 Eigen::MatrixXd V(n_bases, dim);
531 for (const auto &bs : space_.basis_list())
532 for (const auto &b : bs.bases)
533 for (const auto &g : b.global())
534 V.row(g.index) = g.node;
535
536 Eigen::MatrixXi primitives = dim == 2 ? collision_mesh_.edges() : collision_mesh_.faces();
537 for (int entry = 0; entry < primitives.size(); ++entry)
538 {
539 primitives(entry) = collision_mesh_.to_full_vertex_id(primitives(entry));
540 if (primitives(entry) < 0 || primitives(entry) >= n_bases)
541 {
542 log_and_throw_error("Periodic contact requires collision vertices to map to FE basis nodes.");
543 }
544 }
545
546 Eigen::MatrixXd bbox(V.cols(), 2);
547 bbox.col(0) = V.colwise().minCoeff();
548 bbox.col(1) = V.colwise().maxCoeff();
549
550 // Exclude artificial cell cuts, not physical faces whose vertices lie on different periodic boundaries.
551 {
552 std::vector<int> ind;
553 for (int primitive = 0; primitive < primitives.rows(); ++primitive)
554 {
555 int common_faces = periodic_face_mask(primitives(primitive, 0));
556 for (int corner = 1; corner < primitives.cols(); ++corner)
557 {
558 common_faces &= periodic_face_mask(primitives(primitive, corner));
559 }
560 if (common_faces == 0)
561 {
562 ind.push_back(primitive);
563 }
564 }
565
566 primitives = primitives(ind, Eigen::all).eval();
567 }
568
569 Eigen::MatrixXd Vtmp(V.rows() * tile_count, dim);
570 Eigen::MatrixXi tiled_primitives(primitives.rows() * tile_count, primitives.cols());
571
572 if (periodic_tile_offsets.rows() != dim || periodic_tile_offsets.cols() != dim
573 || Eigen::FullPivLU<Eigen::MatrixXd>(periodic_tile_offsets).rank() != dim)
574 log_and_throw_error("Periodic contact requires {} linearly independent periodic boundary pairs", dim);
575 const Eigen::MatrixXd &tile_offset = periodic_tile_offsets;
576
577 for (int tile = 0; tile < tile_count; ++tile)
578 {
579 Eigen::VectorXd block_id(dim);
580 for (int axis = 0; axis < dim; ++axis)
581 {
582 block_id(axis) = (tile >> (dim - axis - 1)) % n_tiles;
583 }
584 Vtmp.middleRows(tile * V.rows(), V.rows()) = V.rowwise() + (tile_offset * block_id).transpose();
585 tiled_primitives.middleRows(tile * primitives.rows(), primitives.rows()) = primitives.array() + tile * V.rows();
586 }
587
588 // clean duplicated vertices
589 Eigen::VectorXi indices;
590 {
591 std::vector<int> tmp;
592 for (int i = 0; i < V.rows(); i++)
593 {
594 if (periodic_dof_mask(i))
595 tmp.push_back(i);
596 }
597
598 indices.resize(tmp.size() * tile_count);
599 for (int i = 0; i < tile_count; i++)
600 {
601 indices.segment(i * tmp.size(), tmp.size()) = Eigen::Map<Eigen::VectorXi, Eigen::Unaligned>(tmp.data(), tmp.size());
602 indices.segment(i * tmp.size(), tmp.size()).array() += i * V.rows();
603 }
604 }
605
606 Eigen::VectorXi potentially_duplicate_mask(Vtmp.rows());
607 potentially_duplicate_mask.setZero();
608 potentially_duplicate_mask(indices).array() = 1;
609
610 Eigen::MatrixXd candidates = Vtmp(indices, Eigen::all);
611
612 Eigen::VectorXi SVI;
613 std::vector<int> SVJ;
614 SVI.setConstant(Vtmp.rows(), -1);
615 int id = 0;
616 double relative_tolerance = 1e-5;
617 for (const json &condition : conditions)
618 relative_tolerance = std::max(relative_tolerance, condition.value("tolerance", 1e-5));
619 const double eps = (bbox.col(1) - bbox.col(0)).maxCoeff() * relative_tolerance;
620 for (int i = 0; i < Vtmp.rows(); i++)
621 {
622 if (SVI[i] < 0)
623 {
624 SVI[i] = id;
625 SVJ.push_back(i);
626 if (potentially_duplicate_mask(i))
627 {
628 Eigen::VectorXd diffs = (candidates.rowwise() - Vtmp.row(i)).rowwise().norm();
629 for (int j = 0; j < diffs.size(); j++)
630 if (diffs(j) < eps)
631 SVI[indices[j]] = id;
632 }
633 id++;
634 }
635 }
636
637 const Eigen::MatrixXd Vnew = Vtmp(SVJ, Eigen::all);
638 for (int entry = 0; entry < tiled_primitives.size(); ++entry)
639 {
640 tiled_primitives(entry) = SVI(tiled_primitives(entry));
641 }
642 Eigen::MatrixXi Enew;
643 Eigen::MatrixXi boundary_triangles(0, 3);
644 if (dim == 3)
645 {
646 boundary_triangles = std::move(tiled_primitives);
647 igl::edges(boundary_triangles, Enew);
648 }
649 else
650 {
651 Enew = std::move(tiled_primitives);
652 }
653
654 std::vector<bool> is_on_surface = ipc::CollisionMesh::construct_is_on_surface(Vnew.rows(), Enew);
655
656 Eigen::SparseMatrix<double> displacement_map;
657 periodic_collision_mesh_ = ipc::CollisionMesh(is_on_surface,
658 std::vector<bool>(Vnew.rows(), false),
659 Vnew,
660 Enew,
661 boundary_triangles,
662 displacement_map);
663
664 periodic_collision_mesh_.init_area_jacobians();
665
666 periodic_collision_mesh_to_basis_.setConstant(Vnew.rows(), -1);
667 for (int i = 0; i < V.rows(); i++)
668 for (int j = 0; j < tile_count; j++)
669 periodic_collision_mesh_to_basis_(SVI[j * V.rows() + i]) = i;
670
671 if (periodic_collision_mesh_to_basis_.maxCoeff() + 1 != V.rows())
672 log_and_throw_error("Failed to tile mesh!");
673 }

940 Eigen::VectorXd NonlinearElasticVarForm::init_homogenization_solve(
941 Eigen::MatrixXd &solution,
942 const double time,
943 const InitialConditionOverride *initial_condition_override)
944 {
945 assert(has_macro_strain());
946 forms.clear();
947 solve_data_ = solver::SolveData();
948 // Retain the RHS assembler so rebuilding forms also retains the Dirichlet constraints that remove translation modes.
949 solve_data_.rhs_assembler = rhs_assembler_;
950 displacement_gradient_.resize(0, 0);
951 macro_strain_constraint_ = assembler::MacroStrainValue();
952 macro_strain_constraint_.init(
953 mesh_->dimension(), args["constraints"]["macro_displacement_gradient"], resources_);
954 init_solve_data(solution, time, "", initial_condition_override);
955
956 for (const auto &[name, form] : solve_data_.named_forms())
957 {
958 if (name == "augmented_lagrangian")
959 {
960 form->set_weight(0);
961 form->disable();
962 }
963 }
964
965 // See chap 3.4 of https://dl.acm.org/doi/10.1145/3687765.
966 // Since rotation does not affect elastic behavior, enforce symmetry of macro strain tensor G.
967 bool solve_symmetric_macro_strain = false;
968 const Eigen::VectorXi &fixed_entries = macro_strain_constraint_.get_fixed_entry();
969 const int dim = mesh_->dimension();
970 for (int i = 0; i < dim && !solve_symmetric_macro_strain; ++i)
971 {
972 for (int j = 0; j < i; ++j)
973 {
974 const bool ij_fixed = std::find(
975 fixed_entries.data(), fixed_entries.data() + fixed_entries.size(), i + j * dim)
976 != fixed_entries.data() + fixed_entries.size();
977 const bool ji_fixed = std::find(
978 fixed_entries.data(), fixed_entries.data() + fixed_entries.size(), j + i * dim)
979 != fixed_entries.data() + fixed_entries.size();
980 if (!ij_fixed && !ji_fixed)
981 solve_symmetric_macro_strain = true;
982 }
983 }
984
985 double characteristic_length = args["solver"]["advanced"]["characteristic_length"];
986 if (characteristic_length <= 0)
987 {
988 RowVectorNd min, max;
989 mesh_->bounding_box(min, max);
990 characteristic_length = (max - min).norm();
991 }
992 double characteristic_force_density = args["solver"]["advanced"]["characteristic_force_density"];
993 if (characteristic_force_density <= 0)
994 characteristic_force_density = 10000;
995
996 const int ndof = space_.n_bases * dim;
997 auto homo_problem = std::make_shared<solver::NLHomoProblem>(
998 ndof, macro_strain_constraint_, space_.n_bases, space_.mesh_nodes,
999 time, forms, solve_data_.al_form, solve_symmetric_macro_strain,
1000 polysolve::linear::Solver::create(args["solver"]["linear"], logger()),
1001 characteristic_length, characteristic_force_density, pure_mass_, dim);
1002 if (solve_data_.periodic_contact_form)
1003 homo_problem->add_form(solve_data_.periodic_contact_form);
1004 if (solve_data_.strain_al_lagr_form)
1005 homo_problem->add_form(solve_data_.strain_al_lagr_form);
1006
1007 solve_data_.nl_problem = homo_problem;
1008 const Eigen::VectorXd initial_reduced = Eigen::VectorXd::Zero(
1009 homo_problem->reduced_size() + homo_problem->macro_reduced_size());
1010 homo_problem->init(initial_reduced);
1011 homo_problem->update_quantities(time, initial_reduced);
1012 stats.solver_info = json::array();
1013
1014 if (!initial_condition_override || initial_condition_override->displacement_gradient.size() == 0)
1015 {
1016 return Eigen::VectorXd::Zero(ndof + dim * dim);
1017 }
1018
1019 const Eigen::MatrixXd &gradient = initial_condition_override->displacement_gradient;
1020 Eigen::VectorXd extended(ndof + dim * dim);
1021 // The continuation state stores total displacement; the homogenization unknown stores u - G X.
1022 extended.head(ndof) = solution - io::Evaluator::generate_linear_field(space_.n_bases, space_.mesh_nodes, gradient);
1023 extended.tail(dim * dim) = utils::flatten(gradient);
1024 return extended;
1025 }
1026
1027 void NonlinearElasticVarForm::solve_homogenization_step(
1028 const int step,
1029 const double time,
1030 Eigen::VectorXd &extended_solution,
1031 Eigen::MatrixXd &solution,
1032 const ForwardStepCallback &post_step)
1033 {
1034
1035 // Approach the prescribed macro strain with AL, then eliminate fixed macro components to enforce it exactly.
1036
1037 auto homo_problem = std::dynamic_pointer_cast<solver::NLHomoProblem>(solve_data_.nl_problem);
1038 assert(homo_problem && solve_data_.strain_al_lagr_form);
1039
1040 const int dim = mesh_->dimension();
1041 if (extended_solution.size() == 0)
1042 extended_solution.setZero(homo_problem->full_size() + dim * dim);
1043 assert(extended_solution.size() == homo_problem->full_size() + dim * dim);
1044 const Eigen::VectorXi &fixed_entries = macro_strain_constraint_.get_fixed_entry();
1045 homo_problem->set_fixed_entry({});
1046
1047 auto lagrangian_form = solve_data_.strain_al_lagr_form;
1048 lagrangian_form->enable();
1049 Eigen::VectorXd reduced_solution = homo_problem->extended_to_reduced(extended_solution);
1050 homo_problem->update_quantities(time, reduced_solution);
1051 reduced_solution = homo_problem->extended_to_reduced(extended_solution);
1052 const Eigen::VectorXd initial_solution = reduced_solution;
1053 const Eigen::VectorXi fixed_indices = fixed_entries.array() + homo_problem->full_size();
1054 const Eigen::VectorXd fixed_values =
1055 utils::flatten(macro_strain_constraint_.eval(time))(fixed_entries);
1056 const double initial_error = lagrangian_form->compute_error(extended_solution);
1057
1058 double al_weight = args["solver"]["augmented_lagrangian"]["initial_weight"];
1059 const double max_weight = args["solver"]["augmented_lagrangian"]["max_weight"];
1060 const double eta_tolerance = args["solver"]["augmented_lagrangian"]["eta"];
1061 const double scaling = args["solver"]["augmented_lagrangian"]["scaling"];
1062 lagrangian_form->set_initial_weight(al_weight);
1063 homo_problem->init(reduced_solution);
1064 homo_problem->solution_changed(reduced_solution);
1065
1066 // Adjust the initial AL weight so that negative-gradient descent points toward the prescribed macro strain.
1067 while (initial_error > 0 && al_weight < max_weight && scaling > 1)
1068 {
1069 Eigen::VectorXd gradient;
1070 homo_problem->gradient(reduced_solution, gradient);
1071 const Eigen::VectorXd macro_gradient = utils::flatten(homo_problem->reduced_to_disp_grad(gradient, true));
1072 if ((macro_gradient(fixed_entries).array() * (extended_solution(fixed_indices) - fixed_values).array()).minCoeff() >= 0)
1073 break;
1074 al_weight = std::min(al_weight * scaling, max_weight);
1075 lagrangian_form->set_initial_weight(al_weight);
1076 }
1077 logger().debug("Macro-strain AL initial penalty: {}", al_weight);
1078
1079 // Try enforcing the prescribed macro strain exactly to check whether the projection is feasible.
1080 extended_solution(fixed_indices) = fixed_values;
1081 Eigen::VectorXd constrained_solution = homo_problem->extended_to_reduced(extended_solution);
1082 homo_problem->line_search_begin(reduced_solution, constrained_solution);
1083 bool force_al_solve = true;
1084
1085 // AL supplies a feasible starting point for the exact constrained solve.
1086 // Once projection is feasible, eliminate the fixed macro components rather
1087 // than continuing to minimize a different, softly constrained problem.
1088 while (force_al_solve
1089 || !std::isfinite(homo_problem->value(constrained_solution))
1090 || !homo_problem->is_step_valid(reduced_solution, constrained_solution)
1091 || !homo_problem->is_step_collision_free(reduced_solution, constrained_solution))
1092 {
1093 force_al_solve = false;
1094 homo_problem->line_search_end();
1095 // Apply any penalty increase from the previous iteration before minimizing again.
1096 lagrangian_form->set_initial_weight(al_weight);
1097 homo_problem->init(reduced_solution);
1098 auto nonlinear_solver = polysolve::nonlinear::Solver::create(
1099 args["solver"]["augmented_lagrangian"]["nonlinear"],
1100 args["solver"]["linear"], units.characteristic_length(), logger());
1101 homo_problem->normalize_forms();
1102 nonlinear_solver->minimize(*homo_problem, reduced_solution);
1103
1104 extended_solution = homo_problem->reduced_to_extended(reduced_solution);
1105 const double current_error = lagrangian_form->compute_error(extended_solution);
1106 const double eta = initial_error > 0 ? 1 - std::sqrt(current_error / initial_error) : 1;
1107 logger().debug("Macro-strain AL penalty {}, squared error {}, progress {}", al_weight, current_error, eta);
1108 if (eta < eta_tolerance && al_weight < max_weight)
1109 al_weight = std::min(al_weight * scaling, max_weight);
1110 else
1111 lagrangian_form->update_lagrangian(extended_solution, al_weight);
1112 if (eta <= 0)
1113 reduced_solution = initial_solution;
1114
1115 extended_solution(fixed_indices) = fixed_values;
1116 constrained_solution = homo_problem->extended_to_reduced(extended_solution);
1117 homo_problem->line_search_begin(reduced_solution, constrained_solution);
1118 }
1119 homo_problem->line_search_end();
1120 lagrangian_form->disable();
1121
1122 homo_problem->set_fixed_entry(fixed_entries);
1123 reduced_solution = homo_problem->extended_to_reduced(extended_solution);
1124 homo_problem->init(reduced_solution);
1125 auto nonlinear_solver = polysolve::nonlinear::Solver::create(
1126 args["solver"]["nonlinear"], args["solver"]["linear"],
1127 units.characteristic_length(), logger());
1128 homo_problem->normalize_forms();
1129 nonlinear_solver->minimize(*homo_problem, reduced_solution);
1130
1131 extended_solution = homo_problem->reduced_to_extended(reduced_solution);
1132 displacement_gradient_ = homo_problem->reduced_to_disp_grad(reduced_solution);
1133 solution = homo_problem->reduced_to_full(reduced_solution);
1134 if (post_step)
1135 post_step(step, solution);
1136 }

1138 Eigen::MatrixXd &sol,
1139 const double t,

1307 const int step,
1308 Eigen::MatrixXd &sol,
1309 const bool init_lagging)
1310 {
1311 assert(solve_data_.nl_problem != nullptr && "Nonlinear forms must initialize the nonlinear problem before solving");
1312 solver::NLProblem &nl_problem = *(solve_data_.nl_problem);
1313
1314 assert(sol.size() == rhs_.size());

1317 {

1319 {
1320 POLYFEM_SCOPED_TIMER("Initializing lagging");
1321 nl_problem.init_lagging(sol);
1322 }
1323 logger().info("Lagging iteration 1:");
1324 }
1325
1326 save_subsolve(0, step, sol);
1327
1328 std::shared_ptr<polysolve::nonlinear::Solver> nl_solver =
1329 polysolve::nonlinear::Solver::create(args["solver"]["augmented_lagrangian"]["nonlinear"], args["solver"]["linear"], units.characteristic_length(), logger());
1330
1331 ALSolver al_solver(
1332 solve_data_.al_form,
1333 args["solver"]["augmented_lagrangian"]["initial_weight"],
1334 args["solver"]["augmented_lagrangian"]["scaling"],
1335 args["solver"]["augmented_lagrangian"]["max_weight"],
1336 args["solver"]["augmented_lagrangian"]["eta"],
1337 [&](const Eigen::VectorXd &x) {
1338 this->solve_data_.update_barrier_stiffness(sol);
1339 });
1340
1341 al_solver.post_subsolve = [&](const double al_weight) {
1342 stats.solver_info.push_back(
1343 {{"type", al_weight > 0 ? "al" : "rc"},
1344 {"t", step},
1345 {"info", nl_solver->info()}});
1346 if (al_weight > 0)
1347 stats.solver_info.back()["weight"] = al_weight;
1348 save_subsolve(stats.solver_info.size(), step, sol);
1349 };
1350
1351 Eigen::MatrixXd prev_sol = sol;
1352 al_solver.solve_al(nl_problem, sol,
1353 args["solver"]["augmented_lagrangian"]["nonlinear"], args["solver"]["linear"], units.characteristic_length());
1354
1355 al_solver.solve_reduced(nl_problem, sol,
1356 args["solver"]["nonlinear"], args["solver"]["linear"], units.characteristic_length());
1357
1358 if (args["space"]["advanced"]["count_flipped_els_continuous"])
1359 {
1360 const auto invalidList = utils::count_invalid(mesh_->dimension(), space_.basis_list(), space_.geometry_basis_list(), sol);
1361 logger().debug("Flipped elements (cnt {}) : {}", invalidList.size(), invalidList);
1362 }

1367 for (int lag_i = 1; !lagging_converged; lag_i++)
1368 {

1370

1390 lag_i - 1, lag_i, grad.norm(), lagging_tol, delta_x_norm, 1e-6);
1391 lagging_converged = false;
1392 break;
1393 }

1395 if (lag_i >= nl_problem.max_lagging_iterations())
1396 {
1397 logger().warn(
1398 "Lagging failed to converge with {:d} iteration(s) (grad_norm={:g} tol={:g})",
1399 lag_i, grad.norm(), lagging_tol);
1400 lagging_converged = false;
1401 break;
1402 }
1403
1404 logger().info("Lagging iteration {:d}:", lag_i + 1);
1405 nl_problem.init(sol);
1406 solve_data_.update_barrier_stiffness(sol);
1407 nl_problem.normalize_forms();
1408 nl_solver->minimize(nl_problem, tmp_sol);
1409 nl_problem.finish();
1410 prev_sol = sol;
1411 sol = nl_problem.reduced_to_full(tmp_sol);
1412
1413 stats.solver_info.push_back(
1414 {{"type", "rc"},
1415 {"t", step},
1416 {"lag_i", lag_i},
1417 {"info", nl_solver->info()}});
1418 save_subsolve(stats.solver_info.size(), step, sol);
1419 }
