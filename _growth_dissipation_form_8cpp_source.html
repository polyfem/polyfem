42 double GrowthDissipationForm::max_step_size(const Eigen::VectorXd &x0, const Eigen::VectorXd &x1) const
43 {
44 // Largest alpha in (0, 1] with x0 + alpha (x1 - x0) >= floor. A DOF
45 // already below its floor (only possible from outside data) is ignored
46 // here and rejected by is_step_valid; returning 0 would abort the line
47 // search outright.
48 double alpha = 1.0;
49 for (int i = 0; i < x0.size(); ++i)
50 {
51 const double dx = x1(i) - x0(i);
52 if (dx >= 0 || x0(i) <= floor_(i))
53 continue;
54 alpha = std::min(alpha, (floor_(i) - x0(i)) / dx * 0.99);
55 }
56 return std::max(alpha, 0.0);
57 }
