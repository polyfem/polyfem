137 StiffnessMatrix block_diag(const StiffnessMatrix &a, const StiffnessMatrix &b)
138 {
139 std::vector<Eigen::Triplet<double>> entries;
140 entries.reserve(a.nonZeros() + b.nonZeros());
141
142 for (int k = 0; k < a.outerSize(); ++k)
143 for (StiffnessMatrix::InnerIterator it(a, k); it; ++it)
144 entries.emplace_back(it.row(), it.col(), it.value());
145
146 for (int k = 0; k < b.outerSize(); ++k)
147 for (StiffnessMatrix::InnerIterator it(b, k); it; ++it)
148 entries.emplace_back(a.rows() + it.row(), a.cols() + it.col(), it.value());
149
150 StiffnessMatrix out(a.rows() + b.rows(), a.cols() + b.cols());
151 out.setFromTriplets(entries.begin(), entries.end());
152 out.makeCompressed();
153 return out;

466 // holds one node of each class (i+j+k) mod 4, and the solve converged to

468 // invertibly onto the point values, and the energy, convex in theta

623 {

625 assign_discr_orders(args["space"], growth_space_id_, mesh, growth_orders, growth_ordersq);

791 void GrowthElasticVarForm::assemble_mass_mat(const mesh::Mesh &mesh, const json &args)
792 {
793 mass_.resize(0, 0);
794 pure_mass_.resize(0, 0);
795 growth_mass_.resize(0, 0);
796
797 igl::Timer timer;
798 timer.start();
799 logger().info("Assembling mass mat...");
800
801 mass_assembler_->assemble(mesh.is_volume(), space_.n_bases, space_.basis_list(), space_.geometry_basis_list(), mass_ass_vals_cache_, 0, mass_, true);
802 pure_mass_assembler_->assemble(mesh.is_volume(), space_.n_bases, space_.basis_list(), space_.geometry_basis_list(), pure_mass_ass_vals_cache_, 0, pure_mass_, true);
803 growth_mass_assembler_->assemble(mesh.is_volume(), growth_space_.n_bases, growth_space_.basis_list(), growth_space_.geometry_basis_list(), growth_mass_ass_vals_cache_, 0, growth_mass_, true);
804
805 assert(mass_.size() > 0);
806 avg_mass_ = 0;
807 for (int k = 0; k < mass_.outerSize(); ++k)
808 for (StiffnessMatrix::InnerIterator it(mass_, k); it; ++it)
809 avg_mass_ += it.value();
810 avg_mass_ /= mass_.rows();
811 logger().info("average mass {}", avg_mass_);
812
813 if (args["solver"]["advanced"]["lump_mass_matrix"])
814 {
815 mass_ = utils::lump_matrix(mass_);
816 growth_mass_ = utils::lump_matrix(growth_mass_);
817 }
818
819 // Static problem: the stacked lumped mass only feeds the nonlinear
820 // problem's scaling. Identity on the growth block (no physical mass).
821 stacked_lumped_mass_ = block_diag(
822 pure_mass_.size() > 0 ? pure_mass_ : identity_mass(displacement_ndof()),
823 identity_mass(growth_ndof()));
824

826 timings.assembling_mass_mat_time = timer.getElapsedTime();
827 logger().info(" took {}s", timings.assembling_mass_mat_time);
828
829 stats.nn_zero = stacked_lumped_mass_.nonZeros();
830 stats.num_dofs = stacked_lumped_mass_.rows();
831 stats.mat_size = (long long)stacked_lumped_mass_.rows() * (long long)stacked_lumped_mass_.cols();
832 logger().info("sparsity: {}/{}", stats.nn_zero, stats.mat_size);
833 }

835 const Eigen::MatrixXd &GrowthElasticVarForm::prescribed_growth_field() const
836 {
837 if (prescribed_growth_.size() > 0)
838 return prescribed_growth_;
839
840 const std::string path = args["boundary_conditions"]["growth_nodal_field"];
841 if (path.empty())
842 return prescribed_growth_; // empty: no whole-field prescription
843
844 auto input = resources_.open(path, false);
845 if (!input || !*input)
846 log_and_throw_error("GrowthElasticity: cannot open growth_nodal_field '{}'.", resources_.describe(path));
847 std::istream &in = *input;
848
849 std::vector<double> values;
850 values.reserve(growth_ndof());
851 double v;
852 while (in >> v)
853 values.push_back(v);
854
855 if (int(values.size()) != growth_ndof())
856 log_and_throw_error(
857 "GrowthElasticity: growth_nodal_field '{}' has {} values, expected {} (one per growth node, mesh-vertex order).",
858 path, values.size(), growth_ndof());
859
860 Eigen::MatrixXd field = Eigen::Map<Eigen::VectorXd>(values.data(), values.size());
861
862 // Mesh-vertex (input) -> internal node order. The internal order is NOT
863 // the file order in general, and no augmented-Lagrangian diagnostic can
864 // catch a wrong map here: initial and target both come from this function.
865 const auto &map = growth_space_.space_in_node_to_node;
866 if (map.size() != field.rows())
867 log_and_throw_error(
868 "GrowthElasticity: growth-space node map has {} entries for {} values; "
869 "growth_nodal_field cannot be placed on the right nodes.",
870 map.size(), field.rows());
871 Eigen::MatrixXd remapped(field.rows(), 1);
872 for (int i = 0; i < map.size(); ++i)
873 remapped(map(i)) = field(i);
874 field = remapped;
875
876 if ((field.array() <= 0).any())
877 log_and_throw_error(
878 "GrowthElasticity: growth_nodal_field '{}' contains non-positive theta (min {}).",
879 path, field.minCoeff());
880
881 prescribed_growth_ = field;
882 return prescribed_growth_;
883 }

885 void GrowthElasticVarForm::initial_growth_solution(Eigen::MatrixXd &solution) const
886 {
887 // A whole-field prescription is also the initial solution: initial ==
888 // target makes the AL start error ~0 and the first iterate consistent.
889 const Eigen::MatrixXd &prescribed = prescribed_growth_field();
890 if (prescribed.size() > 0)
891 {
892 solution = prescribed;
893 return;
894 }
895
896 assert(growth_rhs_assembler_ != nullptr);
897 growth_rhs_assembler_->initial_solution(solution);
898
899 // theta = 0 is singular (Fg not invertible); the harmless zero default
900 // of the temperature precedent is fatal here, so an all-zero initial
901 // growth field -- i.e. no initial_conditions entry for the growth
902 // space -- is replaced by the identity, theta = 1, making the
903 // first iterate pure elasticity per the formulation's initialization
904 // convention. A user-prescribed nonzero field is left untouched.
905 if (solution.size() > 0 && (solution.array() == 0.0).all())
906 {
907 logger().info("Growth field initialized to theta = 1 (identity growth).");
908 solution.setOnes();
909 }
910 }

935 void GrowthElasticVarForm::build_forms(Eigen::MatrixXd &solution, const double t)
936 {
937 assert(solution.cols() == 1);
938 assert(solution.rows() == total_ndof());
939
940 stacked_form_ = std::make_shared<solver::StackedForm>();
941 const auto displacement_block = stacked_form_->add_block(displacement_ndof());
942 const auto growth_block = stacked_form_->add_block(growth_ndof());
943
944 Eigen::MatrixXd displacement, growth;
945 split_solution(solution, displacement, growth);
946
947 solve_data_.time_integrator = nullptr;
948
949 init_forms(args, mesh_->dimension(), displacement, t);
950 for (const auto &form : forms)
951 {
952 assert(form);
953 stacked_form_->add(displacement_block, form);
954 }
955 solve_data_.al_form.clear();
956
957 // No growth-block form: theta has no PDE of its own. The growth block's
958 // stiffness is the K_gg quadrant of the coupling form below plus the
959 // augmented-Lagrangian boundary form; with every growth DOF prescribed
960 // the reduced problem contains only displacements.
961
962 assert(growthelastic_assembler_);
963 growth_coupling_form_ = std::make_shared<solver::MixedAssemblerForm>(
964 space_.n_bases, growth_space_.n_bases,
965 space_.basis_list(), growth_space_.basis_list(), space_.geometry_basis_list(),
966 *growthelastic_assembler_, ass_vals_cache_, growth_ass_vals_cache_,
967 t, /*dt=*/0.0, mesh_->is_volume());
968 stacked_form_->add(displacement_block, growth_block, growth_coupling_form_);
969
970 // Solved growth: the rate-independent dissipation on the growth block.
971 // It sits inside stacked_form_, so the stacked gradient's growth rows
972 // are dPi_inc/dtheta (the KKT residual of the active-set loop).
973 dissipation_form_ = nullptr;
974 if (growth_law_.enabled)
975 {
976 assert(growth_node_weights_.size() == growth_ndof());
977 assert(theta_n_.size() == growth_ndof());
978 dissipation_form_ = std::make_shared<solver::GrowthDissipationForm>(
979 growth_node_weights_, theta_n_, growth_law_.theta_floor_fraction);
980 stacked_form_->add(growth_block, dissipation_form_);
981 }
982
983 forms.clear();
984 forms.push_back(stacked_form_);
985 for (const auto &form : forms)
986 form->set_output_dir(output_path);
987
988 solve_data_.al_form.clear();
989 const bool has_growth_pins = growth_law_.enabled && !pinned_growth_nodes_.empty();
990 if (!boundary_.boundary_nodes.empty() || !growth_boundary_.boundary_nodes.empty()
991 || prescribed_growth_field().size() > 0 || has_growth_pins)
992 {
993 auto stacked_al = std::make_shared<solver::StackedAugmentedLagrangianForm>();
994 const auto displacement_al_block = stacked_al->add_block(displacement_block.size());
995 const auto growth_al_block = stacked_al->add_block(growth_block.size());
996
997 if (!boundary_.boundary_nodes.empty())
998 {
999 // Prescribed growth is one static solve; solved growth steps the
1000 // displacement Dirichlet data through pseudo-time.
1001 stacked_al->add(
1002 displacement_al_block,
1003 std::make_shared<solver::BCLagrangianForm>(
1004 displacement_block.size(),
1005 boundary_.boundary_nodes, boundary_.local_boundary, boundary_.local_neumann_boundary,
1006 elastic_boundary_samples(), mass_, *rhs_assembler_,
1007 obstacle.n_vertices() * mesh_->dimension(),
1008 /*is_time_dependent=*/growth_law_.enabled, t));
1009 }
1010
1011 const Eigen::MatrixXd &prescribed = prescribed_growth_field();
1012 if (growth_law_.enabled)
1013 {
1014 // Active-set pins: every pinned growth DOF is Dirichlet at
1015 // theta_n (tabulated targets). The node list is a member because
1016 // BCLagrangianForm keeps a reference to it.
1017 if (!growth_boundary_.boundary_nodes.empty())
1018 log_and_throw_error("growth_law: growth-space dirichlet_boundary entries are not supported "
1019 "(the growth law owns the growth DOFs; tag displacement BCs with fe_space).");
1020 if (has_growth_pins)
1021 stacked_al->add(
1022 growth_al_block,
1023 std::make_shared<solver::BCLagrangianForm>(
1024 growth_block.size(), pinned_growth_nodes_, growth_mass_,
1025 /*obstacle_ndof=*/0, Eigen::MatrixXd(theta_n_)));
1026 }
1027 else if (prescribed.size() > 0)
1028 {
1029 // Whole-field prescription: every growth DOF is Dirichlet with
1030 // tabulated targets (BCLagrangianForm's target constructor; no
1031 // RhsAssembler involved). Supersedes any growth-space surface
1032 // Dirichlet entries.
1033 if (!growth_boundary_.boundary_nodes.empty())
1034 logger().warn("growth_nodal_field supersedes {} growth-space dirichlet_boundary node(s).", growth_boundary_.boundary_nodes.size());
1035
1036 std::vector<int> all_growth_nodes(growth_ndof());
1037 for (int i = 0; i < growth_ndof(); ++i)
1038 all_growth_nodes[i] = i;
1039
1040 stacked_al->add(
1041 growth_al_block,
1042 std::make_shared<solver::BCLagrangianForm>(
1043 growth_block.size(), all_growth_nodes, growth_mass_,
1044 /*obstacle_ndof=*/0, prescribed));
1045 }
1046 else if (!growth_boundary_.boundary_nodes.empty())
1047 {
1048 assert(growth_space_.disc_orders.size() > 0 && "Growth boundary quadrature requires initialized FE orders");
1049 const int gdiscr_order = mesh_->orders().size() <= 0 ? 1 : mesh_->orders().maxCoeff();
1050 const QuadratureOrders growth_boundary_samples =
1051 n_boundary_samples(growth_space_.disc_orders.maxCoeff(), growth_space_.disc_ordersq.maxCoeff(), gdiscr_order);
1052
1053 stacked_al->add(
1054 growth_al_block,
1055 std::make_shared<solver::BCLagrangianForm>(
1056 growth_block.size(),
1057 growth_boundary_.boundary_nodes, growth_boundary_.local_boundary,
1058 growth_boundary_.local_neumann_boundary, growth_boundary_samples,
1059 growth_mass_, *growth_rhs_assembler_,
1060 /*obstacle_ndof=*/0, /*is_time_dependent=*/false, t));
1061 }
1062
1063 solve_data_.al_form.push_back(stacked_al);
1064 }

1067 void GrowthElasticVarForm::solve_nonlinear_step(const int step, Eigen::MatrixXd &solution, const double grad_tol_factor)
1068 {
1069 assert(solve_data_.nl_problem != nullptr && "Growth forms must initialize the nonlinear problem before solving");
1070 solver::NLProblem &nl_problem = *solve_data_.nl_problem;
1071
1072 json nonlinear_params = solver_params_for_residual_mode(args["solver"]["nonlinear"], nl_problem.is_residual());
1073 if (grad_tol_factor != 1.0 && nonlinear_params.contains("grad_norm_tol"))
1074 nonlinear_params["grad_norm_tol"] = nonlinear_params["grad_norm_tol"].get<double>() * grad_tol_factor;
1075 json al_nonlinear_params = solver_params_for_residual_mode(args["solver"]["augmented_lagrangian"]["nonlinear"], nl_problem.is_residual());
1076 // The AL subsolves run on the AL settings (as in the plain elastic VarForm);
1077 // the main Newton residual tolerance is carried over when it is looser, since
1078 // the penalty phase is where the gradient is largest (absolute test).
1079 if (nonlinear_params.contains("Newton") && nonlinear_params["Newton"].contains("residual_tolerance"))
1080 {
1081 const double main_tol = nonlinear_params["Newton"]["residual_tolerance"];
1082 const double al_tol = al_nonlinear_params.value("Newton", json::object()).value("residual_tolerance", 0.0);
1083 if (main_tol > al_tol)
1084 al_nonlinear_params["Newton"]["residual_tolerance"] = main_tol;
1085 }
1086
1087 std::shared_ptr<polysolve::nonlinear::Solver> nl_solver =
1088 polysolve::nonlinear::Solver::create(
1089 nonlinear_params, args["solver"]["linear"],
1090 units.characteristic_length(), logger());
1091
1092 if (nl_problem.uses_lagging())
1093 nl_problem.init_lagging(solution);
1094
1095 const auto update_displacement_barrier_stiffness = [&](const Eigen::VectorXd &x) {
1096 const Eigen::VectorXd displacement = x.head(displacement_ndof());
1097 solve_data_.update_barrier_stiffness(displacement);
1098 };
1099
1100 if (!solve_data_.al_form.empty())
1101 {
1102 solver::ALSolver al_solver(
1103 solve_data_.al_form,
1104 args["solver"]["augmented_lagrangian"]["initial_weight"],
1105 args["solver"]["augmented_lagrangian"]["scaling"],
1106 args["solver"]["augmented_lagrangian"]["max_weight"],
1107 args["solver"]["augmented_lagrangian"]["eta"],
1108 update_displacement_barrier_stiffness);
1109
1110 al_solver.post_subsolve = [&](const double al_weight) {
1111 stats.solver_info.push_back(
1112 {{"type", al_weight > 0 ? "al" : "rc"},
1113 {"t", step},
1114 {"info", nl_solver->info()}});
1115 if (al_weight > 0)
1116 stats.solver_info.back()["weight"] = al_weight;
1117 save_subsolve(stats.solver_info.size(), step, solution);
1118 };
1119
1120 al_solver.solve_al(
1121 nl_problem, solution, al_nonlinear_params,
1122 args["solver"]["linear"], units.characteristic_length());
1123 al_solver.solve_reduced(
1124 nl_problem, solution, nonlinear_params,
1125 args["solver"]["linear"], units.characteristic_length(), nl_solver);
1126 return;
1127 }

1287 void GrowthElasticVarForm::assemble_growth_node_weights()
1288 {
1289 const int n_el = mesh_->n_elements();
1290
1291 // w_a = sum_e vn_e phi_e int_e N_a dV (growth basis, the element's own
1292 // quadrature), and the plain nodal volume int N_a dV for tolerances.
1293 growth_node_volume_.setZero(growth_ndof());
1294 growth_node_vn_volume_.setZero(growth_ndof());
1295 growth_node_weights_.setZero(growth_ndof());
1296 assembler::ElementAssemblyValues vals;
1297 for (int e = 0; e < n_el; ++e)
1298 {
1299 growth_ass_vals_cache_.compute(
1300 e, mesh_->is_volume(), growth_space_.basis_list()[e], growth_space_.geometry_basis_list()[e], vals);
1301 const Eigen::ArrayXd da = vals.det.array() * vals.quadrature.weights.array();
1302 const double coeff = normal_growth_per_element_(e) * threshold_per_element_(e);
1303 for (const auto &bv : vals.basis_values)
1304 {
1305 const double integral = (bv.val.array() * da).sum();
1306 for (const auto &g : bv.global)
1307 {
1308 growth_node_volume_(g.index) += g.val * integral;
1309 growth_node_vn_volume_(g.index) += g.val * integral * normal_growth_per_element_(e);
1310 growth_node_weights_(g.index) += g.val * integral * coeff;
1311 }
1312 }
1313 }
1314
1315 if (!setup_logged_)
1316 {
1317 const std::string source = growth_law_.from_state
1318 ? fmt::format("from the state (theta_crit {:g}, relag cap {})", growth_law_.theta_crit,
1319 growth_law_.relag_max_updates > 0 ? std::to_string(growth_law_.relag_max_updates) : "none")
1320 : fmt::format("given, in [{:g}, {:g}] Pa", threshold_per_element_.minCoeff(),
1321 threshold_per_element_.maxCoeff());
1322 logger().info("growth_law: threshold {} over {} elements; {} increments to t = {:g}; "
1323 "{} growth quadrature points on element 0.",
1324 source, n_el, growth_law_.n_steps, growth_law_.t_end, vals.quadrature.weights.size());
1325 setup_logged_ = true;
1326 }

1329 void GrowthElasticVarForm::evaluate_threshold_state(
1330 const Eigen::VectorXd &u, const double t,
1331 Eigen::VectorXd &phi_full, Eigen::VectorXd &phi_no_shear, Eigen::VectorXd &theta_tot_out)
1332 {
1333 const auto ge = std::dynamic_pointer_cast<assembler::GrowthElasticity>(growthelastic_assembler_);
1334 if (!ge)
1335 log_and_throw_error("growth_law: the C_check threshold needs the GrowthElasticity assembler.");
1336 const int n_el = mesh_->n_elements();
1337 const int dim = mesh_->dimension();
1338 if (dim != 3)
1339 log_and_throw_error("growth_law: the C_check threshold is 3D only.");
1340 if (!(growth_law_.theta_crit > 0))
1341 log_and_throw_error("growth_law: the C_check threshold needs theta_crit > 0.");
1342 const double crit = growth_law_.theta_crit;
1343 phi_full.resize(n_el);
1344 phi_no_shear.resize(n_el);
1345 theta_tot_out.resize(n_el);
1346
1347 assembler::ElementAssemblyValues vals;
1348 for (int e = 0; e < n_el; ++e)
1349 {
1350 ass_vals_cache_.compute(e, mesh_->is_volume(), space_.basis_list()[e], space_.geometry_basis_list()[e], vals);
1351 const Eigen::ArrayXd da = vals.det.array() * vals.quadrature.weights.array();
1352 Eigen::Matrix3d F = Eigen::Matrix3d::Zero();
1353 RowVectorNd pos = RowVectorNd::Zero(dim);
1354 for (int p = 0; p < da.size(); ++p)
1355 {
1356 Eigen::Matrix3d Fp = Eigen::Matrix3d::Identity();
1357 for (const auto &bv : vals.basis_values)
1358 for (const auto &g : bv.global)
1359 for (int d = 0; d < dim; ++d)
1360 Fp.row(d) += g.val * u(g.index * dim + d) * bv.grad_t_m.row(p);
1361 F += da(p) * Fp;
1362 pos += da(p) * vals.val.row(p);
1363 }
1364 F /= da.sum();
1365 pos /= da.sum();
1366 const Eigen::Matrix3d C = F.transpose() * F;
1367
1368 // wall frame: n0 and any orthonormal in-plane pair
1369 const Eigen::Vector3d n = ge->wall_normal(e).normalized();
1370 const Eigen::Vector3d trial = std::abs(n(0)) < 0.9 ? Eigen::Vector3d::UnitX() : Eigen::Vector3d::UnitY();
1371 const Eigen::Vector3d e1 = (trial - trial.dot(n) * n).normalized();
1372 const Eigen::Vector3d e2 = n.cross(e1);
1373 const double c11 = e1.dot(C * e1), c22 = e2.dot(C * e2), c12 = e1.dot(C * e2);
1374 const double theta_tot = std::sqrt(std::max(c11 * c22 - c12 * c12, 0.0));
1375 if (!(theta_tot > 0))
1376 log_and_throw_error("growth_law: degenerate in-plane stretch on element {}.", e);
1377 theta_tot_out(e) = theta_tot;
1378
1379 // C_check: Fg at theta = theta_tot / theta_crit (so vartheta^e = theta_crit), vn_e
1380 const double th = theta_tot / crit;
1381 const double vn = normal_growth_per_element_(e);
1382 const Eigen::Matrix3d nn = n * n.transpose();
1383 const Eigen::Matrix3d P = Eigen::Matrix3d::Identity() - nn;
1384 const Eigen::Matrix3d Fgi = P / std::sqrt(th) + nn / vn;
1385 const Eigen::Matrix3d Ce = Fgi.transpose() * C * Fgi;
1386
1387 // shear-free state: transverse shears removed, Ce_nn at the layer value
1388 const double layer = 1.0 / n.dot(Ce.inverse() * n);
1389 const Eigen::Matrix3d Ce_ns = P * Ce * P + layer * nn;
1390
1391 phi_full(e) = ge->driving_force(e, pos, t, Ce);
1392 phi_no_shear(e) = ge->driving_force(e, pos, t, Ce_ns);
1393 }
1394 if (!phi_full.allFinite() || !phi_no_shear.allFinite())
1395 log_and_throw_error("growth_law: non-finite C_check threshold.");
1396 }

1450 const Eigen::VectorXd &mult, const std::vector<char> &pinned)
1451 {
1452 // dPi/dtheta per v_n-weighted nodal volume is phi - f_I in Pa (nodal average):
1453 // ~0 on free DOFs (the law holds), >= 0 on pinned ones (the slack).
1454 std::vector<double> free_res;

1465 }
1466 else

1468 }
1469 double med = 0, mx = 0;
1470 if (!free_res.empty())
1471 {
1472 mx = *std::max_element(free_res.begin(), free_res.end());

1483 void GrowthElasticVarForm::accumulate_dissipation(const int step, const double t, const Eigen::VectorXd &dtheta)
1484 {
1485 // Split of the increment's dissipation term v_n phi dtheta (per reference
1486 // volume, Pa) for the from_state threshold: D = v_n max(f_par, 0) dtheta >= 0,
1487 // the dissipation, and S = v_n f_gamma dtheta, the shear supply. A
1488 // shear supply comparable to D means the shear credit carries the growth.
1489 const int n_el = mesh_->n_elements();
1490 if (threshold_parallel_.size() != n_el || shear_supply_.size() != n_el)
1491 return;
1492 if (element_volume_.size() != n_el)
1493 {
1494 element_volume_.resize(n_el);
1495 assembler::ElementAssemblyValues vals;
1496 for (int e = 0; e < n_el; ++e)
1497 {
1498 growth_ass_vals_cache_.compute(
1499 e, mesh_->is_volume(), growth_space_.basis_list()[e], growth_space_.geometry_basis_list()[e], vals);
1500 element_volume_(e) = (vals.det.array() * vals.quadrature.weights.array()).sum();
1501 }
1502 dissipation_.setZero(n_el);
1503 shear_work_.setZero(n_el);
1504 }
1505 const Eigen::VectorXd dth = element_mean_growth(dtheta.cwiseMax(0.0));
1506 const Eigen::VectorXd D = normal_growth_per_element_.cwiseProduct(threshold_parallel_).cwiseProduct(dth);
1507 const Eigen::VectorXd S = normal_growth_per_element_.cwiseProduct(shear_supply_).cwiseProduct(dth);
1508 dissipation_ += D;
1509 shear_work_ += S;
1510 const double iD = D.dot(element_volume_), iS = S.dot(element_volume_);
1511 const double cD = dissipation_.dot(element_volume_), cS = shear_work_.dot(element_volume_);
1512 int n_grow = 0, n_neg = 0, n_dneg = 0;
1513 for (int e = 0; e < n_el; ++e)
1514 if (dth(e) > 0)
1515 {
1516 ++n_grow;
1517 n_neg += (threshold_per_element_(e) < 0);
1518 n_dneg += (D(e) < 0);
1519 }
1520 logger().info("growth step {} (t = {:g}): dissipation int D dV {:.4g} (cumulative {:.4g}), shear supply int S dV "
1521 "{:.4g} (cumulative {:.4g}) [Pa x mesh volume]; S/D {:.3f}; growing elements {}, with phi < 0: {}, with D < 0: {}",
1522 step, t, iD, cD, iS, cS, iD != 0 ? iS / iD : 0.0, n_grow, n_neg, n_dneg);
1523 stats.solver_info.push_back(
1524 {{"type", "growth_dissipation"}, {"t", t}, {"step", step}, {"D", iD}, {"S", iS}, {"D_cumulative", cD}, {"S_cumulative", cS}, {"n_phi_negative", n_neg}, {"n_D_negative", n_dneg}});
1525 }

1527 void GrowthElasticVarForm::report_lag_residual(const int step, const double t, const Eigen::MatrixXd &solution)
1528 {
1529 // The threshold the converged state itself implies (the from_state
1530 // construction at its own C_check) against the one the increment used:
1531 // the lag left by capping or converging the threshold iteration.
1532 Eigen::VectorXd full, ns, tt;
1533 evaluate_threshold_state(solution.col(0).head(displacement_ndof()), t, full, ns, tt);
1534 const Eigen::VectorXd phi_state = ns.cwiseMax(0.0) + (full - ns);
1535 lag_residual_ = phi_state - threshold_per_element_;
1536 const Eigen::VectorXd th_e = element_mean_growth(solution.col(0).tail(growth_ndof()));
1537 attainment_ = tt.cwiseQuotient(th_e) / growth_law_.theta_crit;
1538
1539 std::vector<double> res, att;
1540 for (int e = 0; e < th_e.size(); ++e)
1541 if (th_e(e) > 1.0 + 1e-9 && ns(e) >= 0) // grown, not clamped at this state's C_check
1542 {
1543 res.push_back(std::abs(lag_residual_(e)));
1544 att.push_back(attainment_(e));
1545 }
1546 if (res.empty())
1547 {
1548 logger().info("growth step {} (t = {:g}): lag residual: no grown unclamped elements yet", step, t);
1549 return;
1550 }
1551 const auto q = [](std::vector<double> v, const double f) {
1552 const size_t k = std::min(v.size() - 1, size_t(f * double(v.size() - 1) + 0.5));
1553 std::nth_element(v.begin(), v.begin() + k, v.end());
1554 return v[k];
1555 };
1556 logger().info("growth step {} (t = {:g}): lag residual |phi(state) - phi used| median {:.4g} Pa (p95 {:.4g}); "
1557 "attainment median {:.4f} (p5 {:.4f}, p95 {:.4f}) over {} grown unclamped elements",
1558 step, t, q(res, 0.5), q(res, 0.95), q(att, 0.5), q(att, 0.05), q(att, 0.95), res.size());
1559 stats.solver_info.push_back(
1560 {{"type", "growth_lag_residual"}, {"t", t}, {"step", step}, {"abs_median", q(res, 0.5)}, {"abs_p95", q(res, 0.95)}, {"attainment_median", q(att, 0.5)}});
1561 }

1577
1578 int GrowthElasticVarForm::solve_growth_increment(const int step, const double t, Eigen::MatrixXd &solution)
1579 {
1580 const int nu = displacement_ndof();
1581 const int ng = growth_ndof();
1582
1583 // Release tolerance per DOF: the multiplier is an energy per unit growth
1584 // (Pa x volume), so it is scaled by the nodal volume and the threshold
1585 // magnitude (at least 1 Pa, so a zero threshold still has a scale).
1586
1587 // Pass 1 is the elastic predictor: every growth DOF pinned at theta_n.
1588 // Once the set settles, one confirmation pass re-solves with the gradient
1589 // tolerance scaled by final_tol_factor (when < 1), so the multipliers the
1590 // set was decided on, and the reported state, are converged tighter than
1591 // the release tolerance they are compared with.
1592 std::vector<char> pinned(ng, 1);
1593 bool tight = false;
1594
1595 // from_state: the threshold is re-evaluated on every pass's state, with Aitken
1596 // dynamic relaxation of the iteration: x_{k+1} = x_k + omega_k (G(x_k) - x_k),
1597 // x = (PHI_PAR, f_gamma) per element,
1598 // omega_k = -omega_{k-1} r_{k-1}.(r_k - r_{k-1}) / |r_k - r_{k-1}|^2, clamped.
1599 // Restarted (omega = 1) whenever the active set changed in between, since the
1600 // map G then changes too. The fixed point is the same as without relaxation.
1601 constexpr double omega_min = 0.05, omega_max = 5.0;
1602 Eigen::VectorXd r_prev;
1603 double omega_prev = 1.0;
1604 bool set_changed_last_pass = true;
1605 int n_threshold_updates = 0;
1606 for (int pass = 1; pass <= growth_law_.max_active_set_iterations; ++pass)
1607 {
1608 pinned_growth_nodes_.clear();
1609 for (int a = 0; a < ng; ++a)
1610 {
1611 if (pinned[a])
1612 {
1613 pinned_growth_nodes_.push_back(a);
1614 solution(nu + a, 0) = theta_n_(a); // pinned values exact: AL start error 0
1615 }
1616 }
1617
1618 solve_with_current_pins(step, t, solution, tight ? growth_law_.final_tol_factor : 1.0);
1619
1620 Eigen::VectorXd mult = growth_multiplier(solution);
1621
1622 // from_state: threshold re-evaluated on this pass's state. The dissipation
1623 // is linear in theta with gradient w, so the multiplier for the updated
1624 // threshold is mult + (w_new - w_old), exactly, without another solve.
1625 bool threshold_moved = false;
1626 const bool update_capped = growth_law_.relag_max_updates > 0
1627 && n_threshold_updates >= growth_law_.relag_max_updates;
1628 if (growth_law_.from_state && !update_capped)
1629 {
1630 ++n_threshold_updates;
1631 const Eigen::VectorXd w_old = growth_node_weights_;
1632 const Eigen::VectorXd phi_old = threshold_per_element_;
1633 Eigen::VectorXd par_g, gam_g, clamped_g;
1634 threshold_target(solution, t, par_g, gam_g, clamped_g);
1635 const int n_el = int(par_g.size());
1636 Eigen::VectorXd r(2 * n_el);
1637 r << par_g - threshold_parallel_, gam_g - shear_supply_;
1638 const double dphi = ((par_g + gam_g) - phi_old).cwiseAbs().maxCoeff(); // |G(x) - x|, the lag
1639 double omega = 1.0;
1640 if (!set_changed_last_pass && r_prev.size() == r.size())
1641 {
1642 const Eigen::VectorXd dr = r - r_prev;
1643 const double den = dr.squaredNorm();
1644 if (den > 0)
1645 omega = std::clamp(-omega_prev * r_prev.dot(dr) / den, omega_min, omega_max);
1646 }
1647 r_prev = r;
1648 omega_prev = omega;
1649 set_threshold((threshold_parallel_ + omega * r.head(n_el)).cwiseMax(0.0),
1650 shear_supply_ + omega * r.tail(n_el), clamped_g, t, omega);
1651 mult += growth_node_weights_ - w_old;
1652 // a changed threshold only moves the solution where theta is free
1653 const bool any_free = std::find(pinned.begin(), pinned.end(), 0) != pinned.end();
1654 threshold_moved = any_free && dphi > growth_law_.relag_tol;
1655 const bool cap_now = growth_law_.relag_max_updates > 0
1656 && n_threshold_updates >= growth_law_.relag_max_updates;
1657 logger().info("growth step {} (t = {:g}) pass {}: threshold updated, max |d phi| {:.4g} Pa{}{}",
1658 step, t, pass, dphi, threshold_moved ? " (re-solving)" : "",
1659 cap_now ? fmt::format("; update cap {} reached, threshold held from here", growth_law_.relag_max_updates)
1660 : std::string());
1661 }
1662
1663 // scale from the threshold in use (it may have just been updated)
1664 const double force_scale = std::max(1.0, threshold_per_element_.cwiseAbs().maxCoeff());
1665 const Eigen::VectorXd release_tol = growth_law_.active_set_tol * force_scale * growth_node_volume_;
1666 int n_released = 0, n_pinned = 0, n_pinned_total = 0;
1667 std::vector<char> next = pinned;
1668 for (int a = 0; a < ng; ++a)
1669 {
1670 const double theta = solution(nu + a, 0);
1671 if (pinned[a])
1672 {
1673 // at the bound: the law wants growth when dPi/dtheta < 0
1674 if (mult(a) < -release_tol(a))
1675 {
1676 next[a] = 0;
1677 ++n_released;
1678 }
1679 }
1680 else if (theta < theta_n_(a) - growth_law_.feasibility_tol)
1681 {
1682 next[a] = 1; // free DOF crossed the bound: irreversibility violated
1683 ++n_pinned;
1684 }
1685 n_pinned_total += next[a];
1686 }
1687
1688 growth_pinned_mask_ = pinned;
1689 logger().info("growth step {} (t = {:g}) active-set pass {}: {} released, {} pinned, {}/{} at theta_n",
1690 step, t, pass, n_released, n_pinned, n_pinned_total, ng);
1691 stats.solver_info.push_back(
1692 {{"type", "growth_active_set"}, {"t", t}, {"step", step}, {"pass", pass}, {"released", n_released}, {"pinned", n_pinned}, {"n_at_bound", n_pinned_total}});
1693
1694 set_changed_last_pass = n_released != 0 || n_pinned != 0;
1695 if (n_released == 0 && n_pinned == 0 && !threshold_moved)
1696 {
1697 if (growth_law_.final_tol_factor < 1.0 && !tight)
1698 {
1699 tight = true;
1700 logger().info("growth step {} (t = {:g}): active set settled; confirmation pass with the gradient "
1701 "tolerance x {:g}",
1702 step, t, growth_law_.final_tol_factor);
1703 continue;
1704 }
1705 log_kkt_residual(step, t, solution, mult, pinned);
1706 return pass;
1707 }
1708 pinned = next;
1709 }
1710
1711 log_and_throw_error("growth step {} (t = {:g}): active set did not settle in {} passes "
1712 "(raise growth_law.max_active_set_iterations, or reduce the increment).",
1713 step, t, growth_law_.max_active_set_iterations);
1714 return -1;
1715 }
1716
1717 void GrowthElasticVarForm::solve_growth_law(Eigen::MatrixXd &solution)
1718 {
1719 compute_growth_node_weights();
1720
1721 const int ng = growth_ndof();
1722 theta_n_ = solution.col(0).tail(ng);
1723 if (!(theta_n_.array() > 0).all())
1724 log_and_throw_error("growth_law: initial growth must be positive (min {}).", theta_n_.minCoeff());
1725 growth_activation_step_ = Eigen::VectorXi::Zero(ng);
1726
1727 for (int step = 1; step <= growth_law_.n_steps; ++step)
1728 {
1729 const double t_from = growth_law_.t_end * double(step - 1) / double(growth_law_.n_steps);
1730 const double t = growth_law_.t_end * double(step) / double(growth_law_.n_steps);
1731 const Eigen::VectorXd theta_start = theta_n_;
1732 const int passes = advance_increment(step, t_from, t, solution, 0);
1733
1734 const Eigen::VectorXd theta = solution.col(0).tail(ng);
1735 const Eigen::VectorXd dtheta = theta - theta_start;
1736 const int n_active = int(std::count(growth_pinned_mask_.begin(), growth_pinned_mask_.end(), 0));
1737 logger().info("growth step {}/{} (t = {:g}): {} pass(es), {}/{} DOFs growing, "
1738 "theta in [{:.6g}, {:.6g}], max increment {:.3e}, min increment {:.3e}",
1739 step, growth_law_.n_steps, t, passes, n_active, ng,
1740 theta.minCoeff(), theta.maxCoeff(), dtheta.maxCoeff(), dtheta.minCoeff());
1741 stats.solver_info.push_back(
1742 {{"type", "growth_step"}, {"t", t}, {"step", step}, {"passes", passes}, {"n_growing", n_active}, {"theta_min", theta.minCoeff()}, {"theta_max", theta.maxCoeff()}});
1743
1744 if (growth_law_.theta_crit > 0)
1745 report_lag_residual(step, t, solution);
1746
1747 if (growth_law_.save_increments)
1748 save_growth_increment(step, t, solution);
1749 }
1750 }
1751
1752 int GrowthElasticVarForm::advance_increment(const int step, const double t_from, const double t_to,
1753 Eigen::MatrixXd &solution, const int depth)
1754 {
1755 // Everything an attempt may change, so a failed one can be undone exactly.
1756 const Eigen::MatrixXd solution0 = solution;
1757 const Eigen::VectorXd threshold0 = threshold_per_element_, parallel0 = threshold_parallel_,
1758 shear0 = shear_supply_, clamped0 = threshold_clamped_, weights0 = growth_node_weights_;
1759 const std::vector<char> mask0 = growth_pinned_mask_;
1760 try
1761 {
1762 const int passes = solve_growth_increment(step, t_to, solution);
1763 commit_increment(step, t_to, solution);
1764 return passes;
1765 }
1766 catch (const std::runtime_error &e)
1767 {
1768 if (depth >= growth_law_.max_increment_splits)
1769 throw;
1770 solution = solution0;
1771 threshold_per_element_ = threshold0;
1772 threshold_parallel_ = parallel0;
1773 shear_supply_ = shear0;
1774 threshold_clamped_ = clamped0;
1775 growth_node_weights_ = weights0;
1776 growth_pinned_mask_ = mask0;
1777 const double t_mid = 0.5 * (t_from + t_to);
1778 logger().warn("growth step {}: the increment to t = {:g} failed ({}); retrying from t = {:g} as two "
1779 "half increments, to t = {:g} and t = {:g} (split level {} of {})",
1780 step, t_to, e.what(), t_from, t_mid, t_to, depth + 1, growth_law_.max_increment_splits);
1781 stats.solver_info.push_back(
1782 {{"type", "growth_split"}, {"step", step}, {"t_from", t_from}, {"t_to", t_to}, {"level", depth + 1}});
1783 const int p1 = advance_increment(step, t_from, t_mid, solution, depth + 1);
1784 const int p2 = advance_increment(step, t_mid, t_to, solution, depth + 1);
1785 return p1 + p2;
1786 }
1787 }
1788
1789 void GrowthElasticVarForm::commit_increment(const int step, const double t, Eigen::MatrixXd &solution)
1790 {
1791 const int nu = displacement_ndof();
1792 const int ng = growth_ndof();
1793 const Eigen::VectorXd theta = solution.col(0).tail(ng);
1794 const Eigen::VectorXd dtheta = theta - theta_n_;
1795
1796 accumulate_dissipation(step, t, dtheta);
1797
1798 // Irreversibility holds by construction: pinned DOFs equal theta_n and
1799 // free DOFs passed the feasibility test. Clamp round-off only.
1800 theta_n_ = theta.cwiseMax(theta_n_);
1801 solution.col(0).segment(nu, ng) = theta_n_;
1802
1803 // Activation increment: the first increment in which a DOF
1804 // was free in the settled active set AND grew. Free-but-unmoved
1805 // DOFs (exactly at threshold) do not count.
1806 for (int a = 0; a < ng; ++a)
1807 if (growth_activation_step_(a) == 0 && !growth_pinned_mask_[a] && dtheta(a) > 0)
1808 growth_activation_step_(a) = step;
1809 }

1869 void GrowthElasticVarForm::grown_stress_at(
1870 const int e, const Eigen::MatrixXd &local_pts, const double t,
1871 const Eigen::MatrixXd &displacement, const Eigen::MatrixXd &growth,
1872 Eigen::MatrixXd &cauchy, Eigen::MatrixXd &pk1, Eigen::MatrixXd &pk2,
1873 Eigen::MatrixXd &von_mises, Eigen::MatrixXd &energy) const
1874 {
1875 const auto ge = std::dynamic_pointer_cast<assembler::GrowthElasticity>(growthelastic_assembler_);
1876 if (!ge)
1877 log_and_throw_error("GrowthElastic: grown stresses need the GrowthElasticity assembler.");
1878
1879 const int dim = mesh_->dimension();
1880 const basis::ElementBases &bs = space_.basis_list()[e];
1881 const basis::ElementBases &gbs = space_.geometry_basis_list()[e];
1882 assembler::ElementAssemblyValues vals;
1883 vals.compute(e, mesh_->is_volume(), local_pts, bs, gbs);
1884
1885 Eigen::MatrixXd theta, theta_grad;
1886 io::Evaluator::interpolate_at_local_vals(
1887 *mesh_, 1, growth_space_.basis_list(), growth_space_.geometry_basis_list(),
1888 e, local_pts, growth, theta, theta_grad);
1889
1890 const int n = local_pts.rows();
1891 cauchy.resize(n, 9);
1892 pk1.resize(n, 9);
1893 pk2.resize(n, 9);
1894 von_mises.resize(n, 1);
1895 energy.resize(n, 1);
1896
1897 Eigen::MatrixXd grad_u(dim, dim);
1898 Eigen::Matrix3d s, P, S;
1899 for (int p = 0; p < n; ++p)
1900 {
1901 compute_diplacement_grad(dim, bs, vals, local_pts, p, displacement, grad_u);
1902 const Eigen::Matrix3d F = Eigen::Matrix3d::Identity() + grad_u;
1903 double w;
1904 ge->grown_stresses(e, vals.val.row(p), t, F, theta(p, 0), s, P, S, w);
1905
1906 // column-major, as the base flattening (field "<name>_k" = column k)
1907 cauchy.row(p) = Eigen::Map<const Eigen::Matrix<double, 1, 9>>(s.data());
1908 pk1.row(p) = Eigen::Map<const Eigen::Matrix<double, 1, 9>>(P.data());
1909 pk2.row(p) = Eigen::Map<const Eigen::Matrix<double, 1, 9>>(S.data());
1910 von_mises(p, 0) = von_mises_stress_for_stress_tensor(s);
1911 energy(p, 0) = w;
1912 }
1913 }

1915 void GrowthElasticVarForm::replace_stress_fields(
1916 const io::OutputSample &sample,
1917 const Eigen::MatrixXd &displacement, const Eigen::MatrixXd &growth,
1918 std::vector<io::OutputField> &fields) const
1919 {
1920 if (!mesh_ || mesh_->dimension() != 3 || displacement.size() <= 0 || growth.size() <= 0)
1921 return;
1922 if (!std::dynamic_pointer_cast<assembler::GrowthElasticity>(growthelastic_assembler_))
1923 return;
1924
1925 // Base names: "von_mises" and "<tensor>_k" (k = 1..3, column k of the
1926 // column-major tensor); the nodal averages carry "_avg" before "_k".
1927 // Only fields the base emitted are touched, so the export options keep
1928 // deciding what is written.
1929 static const std::array<std::string, 3> tensor_names = {{"cauchy_stess", "pk1_stess", "pk2_stess"}};
1930 struct Targets
1931 {
1932 io::OutputField *von_mises = nullptr;
1933 std::array<std::array<io::OutputField *, 3>, 3> tensor{};
1934 bool any = false;
1935 };
1936 const auto collect = [&](const std::string &suffix) {
1937 Targets tg;
1938 for (io::OutputField &field : fields)
1939 {
1940 if (field.name == "von_mises" + suffix)
1941 {
1942 tg.von_mises = &field;
1943 tg.any = true;
1944 continue;
1945 }
1946 for (int a = 0; a < 3; ++a)
1947 for (int k = 0; k < 3; ++k)
1948 if (field.name == fmt::format("{}{}_{}", tensor_names[a], suffix, k + 1))
1949 {
1950 tg.tensor[a][k] = &field;
1951 tg.any = true;
1952 }
1953 }
1954 return tg;
1955 };
1956 const Targets point = collect("");
1957 const Targets avg = collect("_avg");
1958 const bool has_element_samples = sample.local_points.rows() > 0 && sample.local_points.rows() == sample.element_ids.size();
1959 Eigen::MatrixXd cauchy, pk1, pk2, vm, w;
1960
1961 // ---- energy density at the output samples; "energy_avg" is the
1962 // per-element mean of the sample values, as in the elastic base
1963 io::OutputField *energy = nullptr, *energy_avg = nullptr;
1964 for (io::OutputField &field : fields)
1965 {
1966 if (field.name == "energy")
1967 energy = &field;
1968 else if (field.name == "energy_avg")
1969 energy_avg = &field;
1970 }
1971 if ((energy || energy_avg) && has_element_samples)
1972 {
1973 const int ns = sample.local_points.rows();
1974 Eigen::VectorXd values = Eigen::VectorXd::Zero(ns);
1975 Eigen::VectorXd sum = Eigen::VectorXd::Zero(mesh_->n_elements());
1976 Eigen::VectorXi count = Eigen::VectorXi::Zero(mesh_->n_elements());
1977 for (int i = 0; i < ns; ++i)
1978 {
1979 const int e = sample.element_ids(i);
1980 if (e < 0)
1981 continue;
1982 grown_stress_at(e, sample.local_points.row(i), sample.time, displacement, growth, cauchy, pk1, pk2, vm, w);
1983 values(i) = w(0, 0);
1984 sum(e) += values(i);
1985 count(e)++;
1986 }
1987 for (int i = 0; i < ns; ++i)
1988 {
1989 const int e = sample.element_ids(i);
1990 if (e < 0)
1991 continue;
1992 if (energy && i < energy->values.rows())
1993 energy->values(i, 0) = values(i);
1994 if (energy_avg && i < energy_avg->values.rows())
1995 energy_avg->values(i, 0) = sum(e) / count(e);
1996 }
1997 }
1998
1999 if (!point.any && !avg.any)
2000 return;
2001
2002 // row `row` of the targets from row `src` of (cauchy, pk1, pk2, von_mises)
2003 const auto write_row = [](const Targets &tg, const int row,
2004 const std::array<const Eigen::MatrixXd *, 3> &tensors,
2005 const Eigen::MatrixXd &vm, const int src) {
2006 if (tg.von_mises && row < tg.von_mises->values.rows())
2007 tg.von_mises->values(row, 0) = vm(src, 0);
2008 for (int a = 0; a < 3; ++a)
2009 for (int k = 0; k < 3; ++k)
2010 if (tg.tensor[a][k] && row < tg.tensor[a][k]->values.rows())
2011 tg.tensor[a][k]->values.row(row) = tensors[a]->block(src, 3 * k, 1, 3);
2012 };
2013
2014 // ---- point values at the output samples
2015 if (point.any && has_element_samples)
2016 {
2017 for (int i = 0; i < sample.local_points.rows(); ++i)
2018 {
2019 const int e = sample.element_ids(i);
2020 if (e < 0)
2021 continue;
2022 grown_stress_at(e, sample.local_points.row(i), sample.time, displacement, growth, cauchy, pk1, pk2, vm, w);
2023 write_row(point, i, {{&cauchy, &pk1, &pk2}}, vm, 0);
2024 }
2025 }
2026
2027 if (!avg.any)
2028 return;
2029
2030 // ---- nodal averages: same rule as the elastic base (element values at
2031 // the element's Lagrange nodes, weighted by element volume, interface
2032 // nodes only), then sampled at the output points
2033 const int nb = space_.n_bases;
2034 Eigen::VectorXd areas = Eigen::VectorXd::Zero(nb);
2035 std::array<Eigen::MatrixXd, 3> avg_tensor;
2036 for (auto &m : avg_tensor)
2037 m.setZero(nb, 9);
2038 Eigen::MatrixXd avg_vm = Eigen::MatrixXd::Zero(nb, 1);
2039
2040 for (int e = 0; e < int(space_.basis_list().size()); ++e)
2041 {
2042 Eigen::MatrixXd local_pts;
2043 if (mesh_->is_simplex(e))
2044 autogen::p_nodes_3d(space_.disc_orders(e), local_pts);
2045 else if (mesh_->is_cube(e))
2046 autogen::q_nodes_3d(space_.disc_orders(e), local_pts);
2047 else if (mesh_->is_prism(e))
2048 autogen::prism_nodes_3d(space_.disc_orders(e), space_.disc_ordersq(e), local_pts);
2049 else
2050 continue;
2051
2052 const basis::ElementBases &bs = space_.basis_list()[e];
2053 assembler::ElementAssemblyValues vals;
2054 vals.compute(e, mesh_->is_volume(), bs, space_.geometry_basis_list()[e]);
2055 const double area = (vals.det.array() * vals.quadrature.weights.array()).sum();
2056
2057 grown_stress_at(e, local_pts, sample.time, displacement, growth, cauchy, pk1, pk2, vm, w);
2058
2059 for (size_t j = 0; j < bs.bases.size(); ++j)
2060 {
2061 const basis::Basis &b = bs.bases[j];
2062 if (b.global().size() > 1)
2063 continue;
2064 const int index = b.global().front().index;
2065 areas(index) += area;
2066 avg_tensor[0].row(index) += cauchy.row(j) * area;
2067 avg_tensor[1].row(index) += pk1.row(j) * area;
2068 avg_tensor[2].row(index) += pk2.row(j) * area;
2069 avg_vm(index) += vm(j) * area;
2070 }
2071 }
2072
2073 for (int i = 0; i < nb; ++i)
2074 {
2075 if (!(areas(i) > 0))
2076 continue;
2077 for (auto &m : avg_tensor)
2078 m.row(i) /= areas(i);
2079 avg_vm(i) /= areas(i);
2080 }
2081
2082 // stack (cauchy | pk1 | pk2 | vm) per node so one interpolation serves all
2083 Eigen::MatrixXd nodal(nb, 28);
2084 nodal << avg_tensor[0], avg_tensor[1], avg_tensor[2], avg_vm;
2085 const Eigen::VectorXd nodal_flat = utils::flatten(nodal);
2086
2087 const auto write_sampled = [&](const int row, const Eigen::MatrixXd &v) {
2088 cauchy = v.middleCols(0, 9);
2089 pk1 = v.middleCols(9, 9);
2090 pk2 = v.middleCols(18, 9);
2091 write_row(avg, row, {{&cauchy, &pk1, &pk2}}, v.middleCols(27, 1), 0);
2092 };
2093
2094 if (has_element_samples)
2095 {
2096 for (int i = 0; i < sample.local_points.rows(); ++i)
2097 {
2098 const int e = sample.element_ids(i);
2099 if (e < 0)
2100 continue;
2101 Eigen::MatrixXd local_sol, local_grad;
2102 io::Evaluator::interpolate_at_local_vals(
2103 *mesh_, 28, space_.basis_list(), space_.geometry_basis_list(),
2104 e, sample.local_points.row(i), nodal_flat, local_sol, local_grad);
2105 write_sampled(i, local_sol);
2106 }
2107 }
2108 else if (sample.node_ids.size() > 0)
