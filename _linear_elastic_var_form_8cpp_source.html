119 {
120 assert(sol.cols() == 1);

122

141

144 solve_data_.time_integrator = time_integrator::ImplicitTimeIntegrator::construct_time_integrator(args["time"]["integrator"]);
145 solve_data_.inertia_form = std::make_shared<solver::InertiaForm>(mass_, *solve_data_.time_integrator);
146
147 POLYFEM_SCOPED_TIMER("Initialize time integrator");
148
149 Eigen::MatrixXd solution, velocity, acceleration;
150 initial_solution(solution, initial_condition_override);
151 solution.col(0) = sol;

154 assert(velocity.rows() == sol.size());

156 assert(acceleration.rows() == sol.size());

158 {
159 log_and_throw_error(
160 "Incompatible initial-condition history for transient solve: "
161 "solution has {} columns, velocity has {}, acceleration has {}.",
162 solution.cols(), velocity.cols(), acceleration.cols());
163 }
164
165 solve_data_.time_integrator->init(solution, velocity, acceleration, dt);
166 restore_checkpoint_integrator(
167 solve_data_.time_integrator, "/checkpoint/state/primary_integrator", dt,
168 "primary", space_, mesh_->dimension());
169
170 solve_data_.elastic_form->set_weight(solve_data_.time_integrator->acceleration_scaling());
171 solve_data_.body_form->set_weight(solve_data_.time_integrator->acceleration_scaling());
172 }

175 solve_data_.time_integrator = nullptr;

197 void LinearElasticVarForm::solve_transient_linear(Eigen::MatrixXd &sol, const ForwardStepCallback &post_step)
198 {
199 assert(problem->is_time_dependent());
200 assert(rhs_assembler_ != nullptr);
201 assert(solve_data_.time_integrator != nullptr && "Transient linear elasticity requires an initialized time integrator");
202
203 auto solver = polysolve::linear::Solver::create(args["solver"]["linear"], logger());
204 logger().info("{}...", solver->name());
205
206 save_timestep(t0, 0, t0, dt, sol);
207 if (post_step)
208 post_step(0, sol);
209
210 Eigen::MatrixXd current_rhs = rhs_;
211
212 StiffnessMatrix stiffness;
213 build_stiffness_mat(stiffness);
214
215 for (int t = 1; t <= time_steps; ++t)
216 {
217 const double time = t0 + t * dt;
218
219 rhs_assembler_->assemble(mass_assembler_->density(), current_rhs, time);
220 current_rhs *= -1;
221
222 rhs_assembler_->set_bc(
223 std::vector<mesh::LocalBoundary>(), std::vector<int>(), elastic_boundary_samples(),
224 boundary_.local_neumann_boundary, current_rhs, sol, time);
225
226 current_rhs *= solve_data_.time_integrator->acceleration_scaling();
227 current_rhs += mass_ * solve_data_.time_integrator->x_tilde();
228
229 rhs_assembler_->set_bc(
230 boundary_.local_boundary, boundary_.boundary_nodes, elastic_boundary_samples(),
231 std::vector<mesh::LocalBoundary>(), current_rhs, sol, time);
232
233 StiffnessMatrix A = stiffness * solve_data_.time_integrator->acceleration_scaling() + mass_;
234 Eigen::VectorXd b = current_rhs;
235
236 solve_linear_system(solver, A, b, args["output"]["advanced"]["spectrum"].get<bool>() && t == 1, sol);
237 if (post_step)
238 post_step(t, sol);
239
240 solve_data_.time_integrator->update_quantities(sol);
241 save_timestep(time, t, t0, dt, sol);
242 save_elastic_step_state(t0, dt, t, sol, solve_data_.time_integrator.get());
243
244 logger().info("{}/{} t={}", t, time_steps, time);
245 notify_time_step(t, time_steps, t0, dt);
246 }

249 void LinearElasticVarForm::solve_problem(
250 Eigen::MatrixXd &sol,
251 const InitialConditionOverride *initial_condition_override,
252 const ForwardStepCallback &post_step)
253 {
254 assert(
255 (problem->is_time_dependent() || !initial_condition_override
256 || (initial_condition_override->velocity.size() == 0
257 && initial_condition_override->acceleration.size() == 0))
258 && "Static elasticity does not accept initial velocity or acceleration overrides");
259
260 stats.spectrum.setZero();
261
262 igl::Timer timer;
263 timer.start();
264 logger().info("Solving {}", primary_assembler_->name());
265
266 {
267 POLYFEM_SCOPED_TIMER("Setup RHS");
268
269 if (initial_condition_override && initial_condition_override->solution.size() != 0)
270 initial_solution(sol, initial_condition_override);
271 else if (sol.size() <= 0)
272 initial_solution(sol, initial_condition_override);
273
274 if (!problem->is_time_dependent())
275 {
276 if (initial_condition_override && initial_condition_override->solution.size() != 0)
277 assert(sol.cols() == 1 && "Static initial solution override must have exactly one column");
278 else if (sol.cols() != 1)
279 log_and_throw_error("Static elasticity requires exactly one initial solution column.");
280 }
281 if (sol.cols() > 1)
282 sol.conservativeResize(Eigen::NoChange, 1);
283 }
284
285 init_linear_solve(sol, problem->is_time_dependent() ? t0 + dt : 1.0, initial_condition_override);
286
287 if (problem->is_time_dependent())
288 solve_transient_linear(sol, post_step);
289 else
290 solve_static_linear(sol, post_step);
291
292 timer.stop();
293 timings.solving_time = timer.getElapsedTime();
294 logger().info(" took {}s", timings.solving_time);
295 }
