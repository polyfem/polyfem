37 std::string resource_destination(const std::string &logical)
38 {
39 // Canonical identities are already normalized. Keep Windows drive
40 // names as components so dependencies on different drives cannot alias.
41 const size_t start = logical.find_first_not_of('/');
42 if (start == std::string::npos || logical == ".")
43 return "/resources/tree";
44 return "/resources/tree/" + logical.substr(start);

47 template <typename T>
48 std::pair<std::vector<T>, std::vector<long>> pack_ragged(const std::vector<std::vector<T>> &rows)
49 {
50 std::pair<std::vector<T>, std::vector<long>> packed;
51 packed.second.reserve(rows.size() + 1);
52 packed.second.push_back(0);
53 for (const auto &row : rows)
54 {
55 packed.first.insert(packed.first.end(), row.begin(), row.end());
56 packed.second.push_back(packed.first.size());
57 }
58 return packed;
59 }

61 Eigen::MatrixXi pack_padded(const std::vector<std::vector<int>> &rows)
62 {
63 int width = 0;
64 for (const auto &row : rows)
65 width = std::max(width, int(row.size()));
66 Eigen::MatrixXi packed = Eigen::MatrixXi::Constant(rows.size(), width, -1);
67 for (int i = 0; i < rows.size(); ++i)
68 for (int j = 0; j < rows[i].size(); ++j)
69 packed(i, j) = rows[i][j];
70 return packed;
71 }

119 void CheckpointWriter::write_mesh(const std::string &group, const mesh::MeshData &data)
120 {
121 data.validate();
122 write_matrix(group + "/vertices", data.vertices);
123 write_int_matrix(group + "/cells", data.elements);
124 write_attribute(group, "schema_version", mesh::MESH_SCHEMA_VERSION);
125 write_attribute(group, "dimension", data.dimension());
126 write_attribute(group, "mesh_type", "fem");
127 write_attribute(group, "elements_are_ordered", long(data.elements_are_ordered));
128 if (!data.body_ids.empty())
129 write_int_vector(group + "/body_ids", data.body_ids);
130 if (!data.geometry_ids.empty())
131 write_int_vector(group + "/geometry_ids", data.geometry_ids);
132 if (!data.node_ids.empty())
133 write_int_vector(group + "/node_ids", data.node_ids);
134 if (!data.boundary_ids.empty())
135 {
136 write_int_matrix(group + "/boundary_elements", pack_padded(data.boundary_elements));
137 write_int_vector(group + "/boundary_ids", data.boundary_ids);
138 }
139 if (!data.higher_order_connectivity.empty())
140 {
141 const auto packed = pack_ragged(data.higher_order_connectivity);
142 write_matrix(group + "/higher_order_nodes", data.higher_order_nodes);
143 write_int_vector(group + "/higher_order_connectivity", packed.first);
144 write_long_vector(group + "/higher_order_offsets", packed.second);
145 }
146 if (!data.higher_order_weights.empty())
147 {
148 const auto packed = pack_ragged(data.higher_order_weights);
149 write_vector(group + "/higher_order_weights", packed.first);
150 write_long_vector(group + "/higher_order_weight_offsets", packed.second);
151 }
152 if (data.has_polyhedral_topology())
153 {
154 const auto faces = pack_ragged(data.faces);
155 const auto cell_faces = pack_ragged(data.cell_faces);
156 const auto orientations = pack_ragged(data.cell_face_orientations);
157 write_int_vector(group + "/faces", faces.first);
158 write_long_vector(group + "/face_offsets", faces.second);
159 write_int_vector(group + "/cell_faces", cell_faces.first);
160 write_long_vector(group + "/cell_face_offsets", cell_faces.second);
161 write_int_vector(group + "/cell_face_orientations", orientations.first);

164 write_int_vector(group + "/cell_is_hex", is_hex);

169 void CheckpointWriter::embed_resources(const ResourceIO &resources)
170 {
171 // Capture the manifest only once, after the completed step. Input reads
172 // during solver setup (e.g. constraints) must be included as well.
173 const std::vector<std::string> manifest = resources.accessed_resources();
174 write_string("/resources/manifest", json(manifest).dump());
175 impl_->file.deleteLink("/resources/root");
176 write_string("/resources/root", resources.canonical_path(""));
177 const auto *hdf5 = dynamic_cast<const HDF5IO *>(&resources);
178 for (const std::string &canonical : manifest)
179 {
180 const std::string destination = resource_destination(canonical);
181 // A previously copied group already contains its descendants.
182 if (impl_->file.linkExists(destination))
183 continue;
184 if (!resources.exists(canonical))
185 log_and_throw_error("Checkpoint dependency {} no longer exists.", resources.describe(canonical));
186 if (hdf5)
187 {
188 // HDF5 object copying preserves datatype, dataspace, attributes and
189 // group contents without converting numeric datasets to byte arrays.
190 impl_->file.copyLinkFromFile(destination, hdf5->file_path(), hdf5->resolve(canonical));
191 continue;
192 }
193 if (resources.is_group(canonical))
194 {
195 impl_->file.createGroup(destination);
196 continue;
197 }
198 auto input = resources.open(canonical, true);
199 const std::vector<unsigned char> contents{
200 std::istreambuf_iterator<char>(*input), std::istreambuf_iterator<char>()};
201 if (input->bad())
202 log_and_throw_error("Unable to embed dependency {}.", resources.describe(canonical));
203 write_bytes(destination, contents);
204 }
205 }

207 void CheckpointWriter::finalize()
208 {
209 if (finalized_)
210 return;
211 impl_.reset();
212 std::error_code error;
213#ifdef _WIN32
214 // Both paths are in the same directory, so this is a rename with
215 // replacement, never a copy followed by deletion.
216 if (!MoveFileExW(temporary_path_.c_str(), path_.c_str(), MOVEFILE_REPLACE_EXISTING))
217 error = std::error_code(GetLastError(), std::system_category());
218#else
219 fs::rename(temporary_path_, path_, error);
220#endif
221 if (error)
222 log_and_throw_error("Unable to atomically publish checkpoint {}: {}", path_.string(), error.message());
223 finalized_ = true;
224 }

250 log_and_throw_error(
