69 {

72 const std::vector<int> vertices(corner_vertex_.begin() + c0, corner_vertex_.begin() + c0 + d);
73 const std::vector<int> adjacent(corner_adjacent_facet_.begin() + c0, corner_adjacent_facet_.begin() + c0 + d);
74 for (int i = 0; i < d; ++i)
75 {
76 const int i_v = d - 1 - i;
77 // The edge starting at the new corner i is the reversed edge that
78 // started at the old corner before i_v.
79 const int i_f = (i_v == 0) ? d - 1 : i_v - 1;
80 corner_vertex_[c0 + i] = vertices[i_v];

126 void PolygonMesh::connect_facets()
127 {
128 // Chain the corners around each vertex.
129 std::vector<int> next_corner_around_vertex(n_corners(), -1);
130 std::vector<int> v2c(n_vertices(), -1);
131 std::vector<int> c2f(n_corners(), -1);
132 for (int f = 0; f < n_facets(); ++f)
133 {
134 for (int c = corners_begin(f); c < corners_end(f); ++c)
135 {
136 const int v = corner_vertex_[c];
137 next_corner_around_vertex[c] = v2c[v];
138 v2c[v] = c;
139 c2f[c] = f;
140 }
141 }
142
143 // Glue each free corner c1 (edge v1 -> v2) to an earlier corner c2 at v1
144 // whose facet has the opposite edge v2 -> v1.
145 for (int f1 = 0; f1 < n_facets(); ++f1)
146 {
147 for (int c1 = corners_begin(f1); c1 < corners_end(f1); ++c1)
148 {
149 if (corner_adjacent_facet_[c1] != -1)
150 continue;
151 const int v2 = corner_vertex_[next_corner_around_facet(f1, c1)];
152 // The chain only links each corner to earlier ones, so c2 != c1.
153 for (int c2 = next_corner_around_vertex[c1]; c2 != -1; c2 = next_corner_around_vertex[c2])
154 {
155 const int f2 = c2f[c2];
156 const int c3 = prev_corner_around_facet(f2, c2);
157 if (corner_vertex_[c3] == v2)
158 {
159 corner_adjacent_facet_[c1] = f2;
160 corner_adjacent_facet_[c3] = f1;
161 break;
162 }
163 }
164 }
165 }

168 int PolygonMesh::connected_components(std::vector<int> &component) const
169 {
170 int nb_components = 0;
171 component.assign(n_facets(), -1);
172 for (int f = 0; f < n_facets(); ++f)
173 {
174 if (component[f] != -1)
175 continue;
176 std::stack<int> S;
177 S.push(f);
178 component[f] = nb_components;
179 do
180 {
181 const int cur_f = S.top();
182 S.pop();
183 for (int c = corners_begin(cur_f); c < corners_end(cur_f); ++c)
184 {
185 const int adj_f = corner_adjacent_facet_[c];
186 if (adj_f != -1 && component[adj_f] == -1)
187 {
188 S.push(adj_f);
189 component[adj_f] = nb_components;
190 }
191 }
192 } while (!S.empty());
193 nb_components++;
194 }
195 return nb_components;
196 }
