76int run_forward_simulation(State &state)
77{
78 state.load_mesh();
79
80 Eigen::MatrixXd sol;
81 state.solve(sol);
82
83 state.variational_formulation->compute_errors(sol);
84
85 io::OutRuntimeData timings = state.variational_formulation->output_timings();
86 logger().info("total time: {}s", timings.total_time());
87
88 state.variational_formulation->save_json(sol);
89 state.variational_formulation->export_data(sol);
90
91 return EXIT_SUCCESS;
92}

188int forward_simulation(const CLI::App &command_line,
189 const io::ResourceIO &resources,
190 const std::string &output_dir,
191 const unsigned max_threads,
192 const bool is_strict,
193 const bool fallback_solver,
194 const spdlog::level::level_enum &log_level,
195 json &in_args)
196{
197 if (in_args.empty())
198 {
199 logger().error("No input file specified!");
200 return command_line.exit(CLI::RequiredError("--json or --hdf5"));
201 }
202
203 json tmp = json::object();
204 if (has_arg(command_line, "log_level"))
205 tmp["/output/log/level"_json_pointer] = int(log_level);
206 if (has_arg(command_line, "max_threads"))
207 tmp["/solver/max_threads"_json_pointer] = max_threads;
208 if (has_arg(command_line, "output_dir"))
209 tmp["/output/directory"_json_pointer] = std::filesystem::absolute(output_dir);
210 if (has_arg(command_line, "enable_overwrite_solver"))
211 tmp["/solver/linear/enable_overwrite_solver"_json_pointer] = fallback_solver;
212 assert(tmp.is_object());
213 in_args.merge_patch(tmp);
214
215 const std::string checkpoint_path =
216 in_args.contains("/input/checkpoint/path"_json_pointer)
217 ? in_args["/input/checkpoint/path"_json_pointer].get<std::string>()
218 : std::string();
219 if (!checkpoint_path.empty())
220 {
221 const std::filesystem::path path = std::filesystem::path(checkpoint_path).is_absolute()
222 ? std::filesystem::path(checkpoint_path)
223 : resources.host_directory() / checkpoint_path;
224 io::CheckpointReader checkpoint(path.lexically_normal());
225 json continuation = checkpoint.config();
226 continuation["input"]["checkpoint"]["path"] = "";
227 continuation["input"]["checkpoint"]["reorder"] =
228 in_args.contains("/input/checkpoint/reorder"_json_pointer)
229 ? in_args["/input/checkpoint/reorder"_json_pointer].get<bool>()
230 : false;
231 return forward_simulation(
232 command_line, checkpoint, output_dir, max_threads,
233 is_strict, fallback_solver, log_level, continuation);
234 }
235
236 return forward_simulation_with_varform_state(resources, in_args, is_strict);
237}
