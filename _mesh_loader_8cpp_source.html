17 std::vector<std::vector<int>> unpack_connectivity(
18 const std::vector<int> &values,
19 const std::vector<long> &offsets,
20 const io::ResourceIO &resources,
21 const std::string &path)
22 {
23 if (offsets.empty() || offsets.front() != 0 || offsets.back() != long(values.size()))
24 log_and_throw_error("Invalid ragged offsets in {}.", resources.describe(path));
25 std::vector<std::vector<int>> result(offsets.size() - 1);
26 for (size_t i = 0; i + 1 < offsets.size(); ++i)
27 {
28 if (offsets[i] > offsets[i + 1])
29 log_and_throw_error("Offsets in {} are not monotone.", resources.describe(path));
30 result[i].assign(values.begin() + offsets[i], values.begin() + offsets[i + 1]);
31 }
32 return result;
33 }

55 {
56 if (resources_.exists(child_path(group, "v")) || resources_.exists(child_path(group, "c")))

60 if (!resources_.has_attribute(group, "schema_version"))

68 log_and_throw_error(

70 resources_.describe(group), expected_type);

221 SurfaceMesh MeshLoader::load_surface(const std::string &path) const
222 {
223 if (!resources_.exists(path))
224 log_and_throw_error("Surface resource {} does not exist.", resources_.describe(path));
225 SurfaceMesh result;
226 if (!resources_.is_group(path))
227 {
228 const std::filesystem::path materialized = resources_.materialize(path);
229 std::string extension = std::filesystem::path(path).extension().string();
230 std::transform(extension.begin(), extension.end(), extension.begin(), ::tolower);
231 if (extension == ".msh")
232 result = MeshReader::read_msh_surface(materialized);
233 else if (extension == ".obj")
234 result = MeshReader::read_obj_surface(materialized);
235 else if (!MeshReader::read_triangle_surface(materialized, result))
236 result = MeshReader::read_geogram_surface(materialized);
237 return result;
238 }
239
240 validate_group(path, "surface");
241 const std::string vertices_path = child_path(path, "vertices");
242 if (!resources_.exists(vertices_path))
243 log_and_throw_error("Surface mesh group {} requires vertices.", resources_.describe(path));
244 result.vertices = resources_.read_matrix(vertices_path);
245 if (result.vertices.cols() != resources_.read_integer_attribute(path, "dimension"))
246 log_and_throw_error("Invalid surface dimension metadata in {}.", resources_.describe(path));
247 const std::string points_path = child_path(path, "points");
248 if (resources_.exists(points_path))
249 {
250 const std::vector<int> values = resources_.read_int_vector(points_path);
251 result.points = Eigen::Map<const Eigen::VectorXi>(values.data(), values.size());
252 }
253 const std::string edges_path = child_path(path, "edges");
254 if (resources_.exists(edges_path))
255 result.edges = resources_.read_int_matrix(edges_path);
256 const std::string faces_path = child_path(path, "faces");
257 if (resources_.exists(faces_path))
258 result.faces = resources_.read_int_matrix(faces_path);
259 const auto valid = [&](const Eigen::MatrixXi &elements) {
260 return elements.size() == 0 || (elements.minCoeff() >= 0 && elements.maxCoeff() < result.vertices.rows());
261 };
262 if ((result.points.size() && (result.points.minCoeff() < 0 || result.points.maxCoeff() >= result.vertices.rows()))
263 || !valid(result.edges) || !valid(result.faces))
264 log_and_throw_error("Surface connectivity in {} references an invalid vertex.", resources_.describe(path));
265 return result;
266 }
