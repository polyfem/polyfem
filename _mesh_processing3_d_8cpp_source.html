100 Vector3d c1 = points.col(hexes[i][v1]);

102 Vector3d c3 = points.col(hexes[i][v3]);
103

107 hex_minJ = jacobian_value;

110 mq.V_Js[id] = jacobian_value;
111 }
112 mq.H_Js[i] = hex_minJ;
113 mq.ave_Jacobian += hex_minJ;
114 if (mq.min_Jacobian > hex_minJ)

116 }
117 mq.ave_Jacobian /= hexes.size();

119 mq.deviation_Jacobian += (mq.H_Js[i] - mq.ave_Jacobian) * (mq.H_Js[i] - mq.ave_Jacobian);
120 mq.deviation_Jacobian /= hexes.size();

568double MeshProcessing3D::a_jacobian(Vector3d &v0, Vector3d &v1, Vector3d &v2, Vector3d &v3)
569{
570 Matrix3d Jacobian;
571
572 Jacobian.col(0) = v1 - v0;
573 Jacobian.col(1) = v2 - v0;
574 Jacobian.col(2) = v3 - v0;
575
576 double norm1 = Jacobian.col(0).norm();
577 double norm2 = Jacobian.col(1).norm();
578 double norm3 = Jacobian.col(2).norm();
579
580 double scaled_jacobian = Jacobian.determinant();
581 if (std::abs(norm1) < Jacobian_Precision || std::abs(norm2) < Jacobian_Precision || std::abs(norm3) < Jacobian_Precision)
582 {
583 return scaled_jacobian;
584 }
585 scaled_jacobian /= norm1 * norm2 * norm3;
586 return scaled_jacobian;

958void MeshProcessing3D::refine_red_refinement_tet(Mesh3DStorage &M, int iter)
959{
960 for (int i = 0; i < iter; i++)
961 {
962 RefinedMesh M_;
963 M_.type = MeshType::TET;
964
965 vector<int> E2V(M.n_edges());
966
967 M_.points.reserve(M.n_vertices() + M.n_edges());
968 for (int v = 0; v < M.n_vertices(); ++v)
969 M_.points.push_back({{M.points(0, v), M.points(1, v), M.points(2, v)}});
970
971 for (int e = 0; e < M.n_edges(); ++e)
972 {
973 Vector3d center;
974 center.setZero();
975 for (auto vid : M.edge_vs[e])
976 center += M.points.col(vid);
977 center /= M.edge_vs[e].size();
978 M_.points.push_back({{center[0], center[1], center[2]}});
979 E2V[e] = M_.points.size() - 1;
980 }
981
982 // The edge shared by two vertices (the edges around a vertex are sorted).
983 auto shared_edge = [&](int v0, int v1, int &e) -> bool {
984 const auto es0 = M.vertex_es[v0], es1 = M.vertex_es[v1];
985 auto it0 = es0.begin(), it1 = es1.begin();
986 while (it0 != es0.end() && it1 != es1.end())
987 {
988 if (*it0 < *it1)
989 ++it0;
990 else if (*it1 < *it0)
991 ++it1;
992 else
993 {
994 e = *it0;
995 return true;
996 }
997 }
998 return false;
999 };
1000
1001 const auto kernel_of = [&](const std::vector<uint32_t> &vs) {
1002 Vector3d center;
1003 center.setZero();
1004 for (const auto &evid : vs)
1005 for (int j = 0; j < 3; j++)
1006 center[j] += M_.points[evid][j];
1007 center /= vs.size();
1008 return std::array<double, 3>{{center[0], center[1], center[2]}};
1009 };
1010
1011 std::vector<bool> e_flag(M.n_edges(), false);
1012
1013 M_.cells.reserve(8 * M.n_cells());
1014 for (int c = 0; c < M.n_cells(); ++c)
1015 { // 1 --> 8
1016 const auto ele_vs = M.cell_vs[c];
1017 const auto ele_fs = M.cell_fs[c];
1018
1019 for (short i = 0; i < 4; i++)
1020 { // four corners
1021 RefinedMesh::Cell ele_;
1022 ele_.fs.resize(4, -1);
1023 ele_.fs_flag.resize(4, 1);
1024 ele_.hex = false;
1025
1026 ele_.vs.push_back(ele_vs[i]);
1027 for (short j = 0; j < 4; j++)
1028 {
1029 if (j == i)
1030 continue;
1031 int v0 = ele_vs[i], v1 = ele_vs[j];
1032 int e = -1;
1033 if (shared_edge(v0, v1, e))
1034 ele_.vs.push_back(E2V[e]);
1035 }
1036 ele_.kernel = kernel_of(ele_.vs);
1037 M_.cells.push_back(std::move(ele_));
1038 }
1039
1040 // 6 edges
1041 std::vector<int> edges(6);
1042 for (int i = 0; i < 6; i++)
1043 {
1044 int v0 = ele_vs[tet_edges[i][0]], v1 = ele_vs[tet_edges[i][1]];
1045 int e = -1;
1046 if (shared_edge(v0, v1, e))
1047 edges[i] = e;
1048 }
1049 // Split the inner octahedron along its shortest diagonal, i.e. between the midpoints
1050 // of two opposite edges (k, 5 - k). A fixed choice of diagonal makes the tets
1051 // degenerate under repeated refinement, since build_connectivity sorts ele.vs.
1052 int diag = 0;
1053 double min_length = std::numeric_limits<double>::max();
1054 for (int k = 0; k < 3; ++k)
1055 {
1056 const double length = (Map<const Vector3d>(M_.points[E2V[edges[k]]].data()) - Map<const Vector3d>(M_.points[E2V[edges[5 - k]]].data())).squaredNorm();
1057 if (length < min_length)
1058 {
1059 min_length = length;
1060 diag = k;
1061 }
1062 }
1063 int lv0 = E2V[edges[diag]], lv1 = E2V[edges[5 - diag]];
1064 e_flag[edges[diag]] = true;
1065 e_flag[edges[5 - diag]] = true;
1066 for (short i = 0; i < 4; i++)
1067 { // four faces
1068 RefinedMesh::Cell ele_;
1069 ele_.fs.resize(4, -1);
1070 ele_.fs_flag.resize(4, 1);
1071 ele_.hex = false;
1072
1073 ele_.vs.push_back(lv0);
1074 ele_.vs.push_back(lv1);
1075 for (short j = 0; j < 3; j++)
1076 {
1077 int c_e = M.face_es[ele_fs[i]][j];
1078 if (e_flag[c_e])
1079 continue;
1080 ele_.vs.push_back(E2V[c_e]);
1081 }
1082 ele_.kernel = kernel_of(ele_.vs);
1083 M_.cells.push_back(std::move(ele_));
1084 }
1085 e_flag[edges[diag]] = e_flag[edges[5 - diag]] = false;
1086 }
1087
1088 const auto point = [&](const uint32_t v) { return Vector3d(M_.points[v][0], M_.points[v][1], M_.points[v][2]); };
1089
1090 // orient tets
1091 const auto t = M.cell_vs[0];
1092 Vector3d c0 = M.points.col(t[0]);
1093 Vector3d c1 = M.points.col(t[1]);
1094 Vector3d c2 = M.points.col(t[2]);
1095 Vector3d c3 = M.points.col(t[3]);
1096 bool signed_volume = a_jacobian(c0, c1, c2, c3) > 0 ? true : false;
1097
1098 for (auto &ele : M_.cells)
1099 {
1100 c0 = point(ele.vs[0]);
1101 c1 = point(ele.vs[1]);
1102 c2 = point(ele.vs[2]);
1103 c3 = point(ele.vs[3]);
1104 bool sign = a_jacobian(c0, c1, c2, c3) > 0 ? true : false;
1105 if (sign != signed_volume)
1106 std::swap(ele.vs[1], ele.vs[3]);
1107 }
1108
1109 // Fs
1110 std::vector<std::array<uint32_t, 3>> total_fs;
1111 total_fs.reserve(M_.cells.size() * 4);
1112 std::vector<std::tuple<uint32_t, uint32_t, uint32_t, uint32_t, uint32_t, uint32_t>> tempF;
1113 tempF.reserve(M_.cells.size() * 4);
1114 std::array<uint32_t, 3> vs;
1115 for (uint32_t c = 0; c < M_.cells.size(); ++c)
1116 {
1117 const auto &ele = M_.cells[c];
1118 for (int i = 0; i < 4; i++)
1119 {
1120 vs[0] = ele.vs[tet_faces[i][0]];
1121 vs[1] = ele.vs[tet_faces[i][1]];
1122 vs[2] = ele.vs[tet_faces[i][2]];
1123 total_fs.push_back(vs);
1124 std::sort(vs.begin(), vs.end());
1125 tempF.push_back(std::make_tuple(vs[0], vs[1], vs[2], c * 4 + i, c, i));
1126 }
1127 }
1128 std::sort(tempF.begin(), tempF.end());
1129 M_.faces.reserve(tempF.size() / 3);
1130 uint32_t F_num = 0;
1131 for (uint32_t i = 0; i < tempF.size(); ++i)
1132 {
1133 if (i == 0 || (i != 0 && (std::get<0>(tempF[i]) != std::get<0>(tempF[i - 1]) || std::get<1>(tempF[i]) != std::get<1>(tempF[i - 1]) || std::get<2>(tempF[i]) != std::get<2>(tempF[i - 1]))))
1134 {
1135 F_num++;
1136 const auto &fvs = total_fs[std::get<3>(tempF[i])];
1137 M_.faces.emplace_back(fvs.begin(), fvs.end());
1138 }
1139
1140 M_.cells[std::get<4>(tempF[i])].fs[std::get<5>(tempF[i])] = F_num - 1;
1141 }
1142
1143 Mesh3DStorage refined = M_.to_storage();
1144 build_connectivity(refined);
1145 orient_volume_mesh(refined);
1146 build_connectivity(refined);
1147
1148 M = std::move(refined);
1149 }
