80 void InflatedPeriodicShapeVariableToSimulation::update(const Eigen::VectorXd &variables)
81 {
82 // Layout: [cell width, graph position DOFs..., vertex radii..., joint blending...].
83 const Eigen::VectorXd parameters = parametrization_.eval(variables);
84 if (parameters.size() != initial_.size() || !parameters.allFinite() || parameters(0) <= 0)
85 {
86 log_and_throw_adjoint_error("Invalid inflated periodic shape parameters.");
87 }
88 if (parameters.size() == last_parameters_.size() && parameters == last_parameters_)
89 {
90 return;
91 }
92
93 const auto directory = std::filesystem::path(work_directory_) / std::to_string(generation_++);
94 if (!std::filesystem::create_directory(directory))
95 {
96 log_and_throw_adjoint_error("Inflator generation directory already exists: {}", directory.string());
97 }
98
99 // Debugging code
100 // const std::string options_path = (directory / "meshing.json").string();
101 // std::ofstream(options_path) << meshing_.dump(4);
102 // std::ofstream(directory / "parameters.json") << json(std::vector<double>(parameters.data(), parameters.data() + parameters.size())).dump(4);
103
104 // Microstructure is represented as a graph composed of joints, each with its own radius.
105 // Due to historical reason, we use an external inflator program to convert between graph
106 // and periodic mesh. This program also computes the chain rule term for first derivative.
107 const inflator::Mesh mesh = inflator::inflate(wire_, options_, parameters.data() + 1, parameters.size() - 1);
108
109 // Layouts: vertices[vertex * dim + axis], elements[cell * (dim + 1) + corner].
110 // Elements are triangles in 2D and tetrahedra in 3D. Scale each length-2 inflator cell
111 // to physical side lengths [width, 1, ...]; tiling multiplies these lengths by tiles.
112 Eigen::MatrixXd vertices(mesh.vertices.size() / dimension_, dimension_);
113 Eigen::MatrixXi cells(mesh.elements.size() / (dimension_ + 1), dimension_ + 1);
114 for (int vertex = 0; vertex < vertices.rows(); ++vertex)
115 {
116 for (int axis = 0; axis < dimension_; ++axis)
117 {
118 const double cell_length = axis == 0 ? parameters(0) : 1.0;
119 vertices(vertex, axis) = mesh.vertices[vertex * dimension_ + axis] * cell_length / 2;
120 }
121 }
122 for (int cell = 0; cell < cells.rows(); ++cell)
123 {
124 for (int corner = 0; corner < cells.cols(); ++corner)
125 {
126 cells(cell, corner) = mesh.elements[cell * (dimension_ + 1) + corner];
127 }
128 }
129
130 // Jacobian rows are [width, graph parameters...]; columns are interleaved vertex coordinates.
131 shape_jacobian_.setZero(parameters.size(), vertices.size());
132 for (int vertex = 0; vertex < vertices.rows(); ++vertex)
133 {
134 shape_jacobian_(0, dimension_ * vertex) = vertices(vertex, 0) / parameters(0);
135 }
136
137 // Velocities use [graph parameter][vertex][axis] order: normal speed times unit normal,
138 // with zero interior entries.
139 for (int parameter = 1; parameter < parameters.size(); ++parameter)
140 {
141 const std::size_t parameter_offset = (parameter - 1) * mesh.vertices.size();
142 for (int vertex = 0; vertex < vertices.rows(); ++vertex)
143 {
144 for (int axis = 0; axis < dimension_; ++axis)
145 {
146 const Eigen::Index coordinate = dimension_ * vertex + axis;
147 const double velocity = mesh.boundary_shape_velocities[parameter_offset + coordinate];
148 const double cell_length = axis == 0 ? parameters(0) : 1.0;
149 shape_jacobian_(parameter, coordinate) = velocity * cell_length / 2;
150 }
151 }
152 }
153 if (!vertices.allFinite() || !shape_jacobian_.allFinite())
154 {
155 log_and_throw_adjoint_error("Non-finite inflator geometry or shape velocities.");
156 }
157
158 // Regeneration can change vertex count and connectivity, requiring new periodic pairings.
159 periodic_map_ = std::make_unique<PeriodicMeshToMesh>(vertices);
160 periodic_parameters_ = periodic_map_->inverse_eval(utils::flatten(vertices));
161 periodic_jacobian_.setZero(parameters.size(), periodic_parameters_.size());
162 // Layout: [unique normalized vertex coordinates, cell matrix entries].
163 // Divide physical velocities by tiled cell lengths and assign each shared coordinate once.
164 std::vector<bool> visited(periodic_map_->n_periodic_dof(), false);
165 for (int vertex = 0; vertex < vertices.rows(); ++vertex)
166 {
167 const int representative = periodic_map_->full_to_periodic(vertex);
168 if (visited[representative])
169 {
170 continue;
171 }
172
173 visited[representative] = true;
174 for (int axis = 0; axis < dimension_; ++axis)
175 {
176 const double tiled_length = options_.tiles * (axis == 0 ? parameters(0) : 1.0);
177 const auto physical_derivatives = shape_jacobian_.block(1, dimension_ * vertex + axis, parameters.size() - 1, 1);
178 auto periodic_derivatives = periodic_jacobian_.block(1, dimension_ * representative + axis, parameters.size() - 1, 1);
179 periodic_derivatives = physical_derivatives / tiled_length;
180 }
181 }
182 // Width changes only the cell matrix in periodic coordinates: cell_xx = tiles * width.
183 const Eigen::Index cell_matrix_offset = periodic_parameters_.size() - dimension_ * dimension_;
184 periodic_jacobian_(0, cell_matrix_offset) = options_.tiles;
185
186 // File-based replace_mesh requires this MSH round-trip of the in-memory library output.
187 const std::string mesh_path = (directory / "mesh.msh").string();
188 const std::vector<int> body_ids(cells.rows(), 0);
189 const bool is_volume = dimension_ == 3;
190 io::MshWriter::write(mesh_path, vertices, cells, body_ids, is_volume, /*binary=*/true);
191 for (int state = 0; state < varforms_.size(); ++state)
192 {
193 varforms_[state]->replace_mesh(mesh_path);
194 // Clear mesh-dependent solutions and derivatives in place; objectives retain shared references to this cache.
195 *diff_caches_[state] = DiffCache();
196 }
197
198 last_parameters_ = parameters;
199 adjoint_logger().info("Inflated mesh generation {}: {} vertices, {} elements", generation_ - 1, vertices.rows(), cells.rows());
200 }

207 Eigen::VectorXd InflatedPeriodicShapeVariableToSimulation::compute_adjoint_term(const Eigen::VectorXd &variables) const
208 {
209 Eigen::VectorXd result = Eigen::VectorXd::Zero(initial_.size());
210 for (int state = 0; state < varforms_.size(); ++state)
211 {
212 Eigen::VectorXd term;
213 AdjointTools::dJ_periodic_shape_adjoint_term(*varforms_[state],
214 *diff_caches_[state],
215 *periodic_map_,
216 periodic_parameters_,
217 diff_caches_[state]->u(0),
218 get_adjoint_mat(*varforms_[state], *diff_caches_[state], 0),
219 term);
220 // Chain the FEM adjoint through the inflator's geometric shape velocities.
221 result += periodic_jacobian_ * term;
222 }
223
224 return parametrization_.apply_jacobian(result, variables);
225 }
