97 std::vector<Eigen::Triplet<double>> entries;
98 entries.reserve(2 * mixed.nonZeros());
99 for (int k = 0; k < mixed.outerSize(); ++k)
100 for (StiffnessMatrix::InnerIterator it(mixed, k); it; ++it)
101 {
102 entries.emplace_back(it.row(), velocity_ndof_ + it.col(), it.value());
103 entries.emplace_back(velocity_ndof_ + it.col(), it.row(), it.value());
104 }
105
