80 bool facets_match(const StandardCell &c1, int f1, const StandardCell &c2, int f2)
81 {
82 const int nbv = c1.n_facet_vertices(f1);
83 if (c2.n_facet_vertices(f2) != nbv)
84 return false;
85 for (int offset = 0; offset < nbv; ++offset)
86 {
87 bool match = true;
88 for (int v1 = 0; v1 < nbv; ++v1)
89 {
90 const int v2 = (nbv - v1 + offset) % nbv;
91 if (c1.facet_vertex(f1, v1) != c2.facet_vertex(f2, v2))
92 {
93 match = false;
94 break;
95 }
96 }
97 if (match)
98 return true;
99 }
100 return false;
101 }

140 void connect_tets(const std::vector<StandardCell> &cells, const int n_vertices, std::vector<std::array<int, 6>> &adjacent)
141 {
142 // Compact copy of the tets (4 vertices each) for cache-friendly access.
143 std::vector<std::array<int, 4>> tets(cells.size());
144 for (int t = 0; t < cells.size(); ++t)
145 for (int lv = 0; lv < 4; ++lv)
146 tets[t][lv] = cells[t].vertices[lv];
147
148 // Chain tet corners around vertices.
149 std::vector<int> next_tet_corner_around_vertex(4 * tets.size(), -1);
150 std::vector<int> v2c(n_vertices, -1);
151 for (int t = 0; t < tets.size(); ++t)
152 {
153 for (int lv = 0; lv < 4; ++lv)
154 {
155 const int v = tets[t][lv];
156 next_tet_corner_around_vertex[4 * t + lv] = v2c[v];
157 v2c[v] = 4 * t + lv;
158 }
159 }
160
161 for (int t1 = 0; t1 < tets.size(); ++t1)
162 {
163 for (int lf1 = 0; lf1 < 4; ++lf1)
164 {
165 if (adjacent[t1][lf1] != -1)
166 continue;
167 const int *fv = tet_descriptor.facet_vertex[lf1];
168 const int v1 = tets[t1][fv[0]];
169 const int v2 = tets[t1][fv[1]];
170 const int v3 = tets[t1][fv[2]];
171 for (int c2 = v2c[v1]; c2 != -1; c2 = next_tet_corner_around_vertex[c2])
172 {
173 const int t2 = c2 / 4;
174 const int lf2 = find_tet_facet(tets[t2], v3, v2, v1);
175 if (lf2 != -1)
176 {
177 adjacent[t1][lf1] = t2;
178 adjacent[t2][lf2] = t1;
179 break;
180 }
181 }
182 }
183 }

204 std::vector<std::array<int, 6>> connect_cells(const std::vector<StandardCell> &cells, const int n_vertices)
205 {
206 std::array<int, 6> unconnected;
207 unconnected.fill(-1);
208 std::vector<std::array<int, 6>> adjacent(cells.size(), unconnected);
209
210 const bool all_tets = std::all_of(cells.begin(), cells.end(), [](const StandardCell &c) {
211 return c.type == StandardCellType::TET;
212 });
213 if (all_tets)
214 {
215 connect_tets(cells, n_vertices, adjacent);
216 return adjacent;
217 }
218
219 // Chain cells around vertices.
220 std::vector<std::array<int, 8>> next_cell_around_vertex(cells.size());
221 std::vector<int> v2cell(n_vertices, -1);
222 for (int c = 0; c < cells.size(); ++c)
223 {
224 for (int lv = 0; lv < cells[c].descriptor().n_vertices; ++lv)
225 {
226 const int v = cells[c].vertices[lv];
227 next_cell_around_vertex[c][lv] = v2cell[v];
228 v2cell[v] = c;
229 }
230 }
231 const auto next_cell = [&](const int c, const int v) {
232 return next_cell_around_vertex[c][find_cell_vertex(cells[c], v)];
233 };
234
235 // Glue each unconnected facet (c1, lf1) to the first matching facet of
236 // a cell incident to its first vertex.
237 for (int c1 = 0; c1 < cells.size(); ++c1)
238 {
239 for (int lf1 = 0; lf1 < cells[c1].n_facets(); ++lf1)
240 {
241 if (adjacent[c1][lf1] != -1)
242 continue;
243 const int v1 = cells[c1].facet_vertex(lf1, 0);
244 for (int c2 = v2cell[v1]; c2 != -1; c2 = next_cell(c2, v1))
245 {
246 const int lf2 = find_cell_facet(cells[c2], cells[c1], lf1);
247 if (lf2 != -1)
248 {
249 adjacent[c1][lf1] = c2;
250 adjacent[c2][lf2] = c1;
251 break;
252 }
253 }
254 }
255 }
256
257 // Reject non-conforming quad/triangle interfaces.
258 std::vector<std::pair<int, int>> matches;
259 for (int c1 = 0; c1 < cells.size(); ++c1)
260 {
261 if (cells[c1].type == StandardCellType::TET)
262 continue;
263 for (int lf1 = 0; lf1 < cells[c1].n_facets(); ++lf1)
264 {
265 if (cells[c1].n_facet_vertices(lf1) != 4 || adjacent[c1][lf1] != -1)
266 continue;
267
268 matches.clear();
269 for (int lv1 = 0; lv1 < 4; ++lv1)
270 {
271 const int v1 = cells[c1].facet_vertex(lf1, lv1);
272 for (int c2 = v2cell[v1]; c2 != -1; c2 = next_cell(c2, v1))
273 {
274 if (c2 == c1 || cells[c2].type == StandardCellType::HEX)
275 continue;
276 for (int lf2 = 0; lf2 < cells[c2].n_facets(); ++lf2)
277 {
278 if (cells[c2].n_facet_vertices(lf2) == 3
279 && triangular_facet_matches_quad_facet(cells[c2], lf2, cells[c1], lf1))
280 matches.emplace_back(c2, lf2);
281 }
282 }
283 }
284 std::sort(matches.begin(), matches.end());
285 matches.erase(std::unique(matches.begin(), matches.end()), matches.end());
286 if (matches.size() >= 2)
287 log_and_throw_error(
288 "Cell {} facet {}: a quadrilateral facet is glued to {} triangular facets; "
289 "non-conforming quad/triangle interfaces are not supported.",
290 c1, lf1, matches.size());
291 }
292 }
293
294 return adjacent;
295 }
