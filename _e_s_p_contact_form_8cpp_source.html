18 ipc::FaceQuadRule build_quad_rule(const int order)
19 {
20 polyfem::quadrature::Quadrature quad;
21 polyfem::quadrature::TriQuadrature(true).get_quadrature(order, quad);
22
23 ipc::FaceQuadRule rule;
24 rule.reserve(quad.size());
25 for (int i = 0; i < quad.size(); i++)
26 {
27 const double x = quad.points(i, 0);
28 const double y = quad.points(i, 1);
29 // polyfem divides weights by 2 (reference triangle area); multiply
30 // back to match ipc-toolkit's convention of weights summing to 1.
31 const double w = quad.weights(i) * 2.0;
32 rule.push_back({{{1.0 - x - y, x, y}}, w});
33 }
34 return rule;
35 }

51 void verify_vertices_in_quad_rule(const ipc::FaceQuadRule &rule)
52 {
53 for (int v = 0; v < 3; v++)
54 {
55 bool found = false;

69 ipc::ESPParameters init_params(const double dhat, const json &esp_params, const bool skip_obstacles, std::shared_ptr<ipc::Barrier> barrier, const int dim)
70 {
71 const int quadrature_order = esp_params["quadrature_order"];
72 const double dbar_factor = esp_params["dbar_factor"];
73 const bool area_weights = esp_params["area_weights"];

158 void ESPContactForm::post_step(const polysolve::nonlinear::PostStepData &data)
159 {
160 const Eigen::MatrixXd displaced_surface = compute_displaced_surface(data.x);
161
162 // Always requires update_collision_set
163 update_collision_set(displaced_surface);
164
165 const double curr_distance = collision_set_.compute_minimum_distance(collision_mesh_, displaced_surface);
166 if (!std::isinf(curr_distance))
167 {
168 const double ratio = sqrt(curr_distance) / dhat();

170 polyfem::logger().log(log_level, "Minimum distance during solve: {}, dhat: {}", sqrt(curr_distance), dhat());
171 }
172
173 if (data.iter_num == 0)
174 return;
