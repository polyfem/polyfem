205 void StackedForm::second_derivative_unweighted(const Eigen::VectorXd &x, StiffnessMatrix &hessian) const
206 {
207 assert(x.size() == size_);
208
209 using StorageIndex = typename StiffnessMatrix::StorageIndex;
210 std::vector<Eigen::Triplet<double, StorageIndex>> entries;
211
212 for (const Term &term : terms_)
213 {
214 if (!term.form->enabled())
215 continue;
216
217 StiffnessMatrix local_hessian;
218 term.form->second_derivative(gather(x, term), local_hessian);
219 const int local_size = term_size(term);
220 assert(local_hessian.rows() == local_size);
221 assert(local_hessian.cols() == local_size);
222
223 for (int k = 0; k < local_hessian.outerSize(); ++k)
224 {
225 for (StiffnessMatrix::InnerIterator it(local_hessian, k); it; ++it)
226 {
227 entries.emplace_back(
228 StorageIndex(global_index(term, int(it.row()))),
229 StorageIndex(global_index(term, int(it.col()))),
230 it.value());
231 }
232 }
233 }
234
235 hessian.resize(size_, size_);
236 hessian.setFromTriplets(entries.begin(), entries.end());

272 {
